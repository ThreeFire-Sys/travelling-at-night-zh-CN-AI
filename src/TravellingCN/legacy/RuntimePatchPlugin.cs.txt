        private void Awake()
        {
            Log = Logger;
            PluginDirectory = Path.GetDirectoryName(Info.Location) ?? Paths.PluginPath;
            _toggleShortcut = Config.Bind(
                "General",
                "ToggleLanguage",
                new KeyboardShortcut(KeyCode.F9),
                "游戏内即时切换简体中文/英文的按键。");
            _startInChinese = Config.Bind(
                "General",
                "StartInChinese",
                true,
                "启动游戏时是否启用简体中文。");
            _fontScale = Config.Bind(
                "Typography",
                "ChineseFontScale",
                1.00f,
                "中文固定字号倍率；建议保持在 1.00–1.10。自动字号文本不放大。");
            // v1.0 used a global 1.04 multiplier.  It enlarged fixed-layout
            // character-generation text into neighbouring blocks, so migrate
            // that old default to the layout-safe value.  Users may still
            // choose a custom value afterwards.
            if (Mathf.Abs(_fontScale.Value - 1.04f) < 0.0001f)
            {
                _fontScale.Value = 1.00f;
            }
            var diagLogging = Config.Bind(
                "Diagnostics",
                "LogLocalizationBoundaries",
                true,
                "诊断构建：记录各本地化边界到达的原文与漏译字符串到 travellingcn-diag.jsonl。");
            DiagLoggingEnabled = diagLogging.Value;
            DiagLogPath = Path.Combine(PluginDirectory, "travellingcn-diag.jsonl");
            ChineseEnabled = _startInChinese.Value;
            ChineseFontScale = Mathf.Clamp(_fontScale.Value, 1f, 1.10f);
            LoadCatalog();
            InstallChineseFont();
            DiagRecord(
                "Session",
                "start",
                $"patch={PluginVersion}; translations={Translations.Count}; " +
                $"links={LinkTargetTranslations.Count}; chinese={ChineseEnabled}");

            _harmony = new Harmony(PluginGuid);
            var patchedClasses = 0;
            var failedPatchClasses = 0;
            foreach (var patchType in typeof(Plugin)
                .GetNestedTypes(BindingFlags.Public | BindingFlags.NonPublic)
                .Where(type => type.GetCustomAttributes(typeof(HarmonyPatch), false).Length > 0)
                .OrderBy(type => type.FullName, StringComparer.Ordinal))
            {
                try
                {
                    var patchedMethods = _harmony.CreateClassProcessor(patchType).Patch();
                    if (patchedMethods == null || patchedMethods.Count == 0)
                    {
                        failedPatchClasses++;
                        Log.LogError($"运行时补丁 {patchType.Name} 未解析到任何目标方法。");
                        continue;
                    }
                    patchedClasses++;
                    Log.LogDebug($"运行时补丁 {patchType.Name}：{patchedMethods.Count} 个目标。");
                }
                catch (Exception exception)
                {
                    failedPatchClasses++;
                    // A game update may change one optional rendering method.
                    // Keep every unrelated localisation boundary alive instead
                    // of allowing Harmony's assembly-wide PatchAll to abort.
                    Log.LogError($"安装运行时补丁 {patchType.Name} 失败：{exception}");
                }
            }
            Log.LogInfo($"运行时补丁类：成功 {patchedClasses}，失败 {failedPatchClasses}。");
            RefreshAll("awake");
            SceneManager.sceneLoaded += OnSceneLoaded;
            StartCoroutine(RefreshLoop());
            Log.LogInfo(
                $"已载入 {Translations.Count} 条简体中文翻译、" +
                $"{LinkTargetTranslations.Count} 个脚注链接译名。" +
                $"当前语言：{(ChineseEnabled ? "简体中文" : "English")}；F9 可即时切换。");
        }

        private void Update()
        {
            if (_toggleShortcut != null && _toggleShortcut.Value.IsDown())
            {
                ToggleLanguage();
            }

        private void OnGUI()
        {
            if (string.IsNullOrEmpty(_toggleNotice) || Time.realtimeSinceStartup > _toggleNoticeUntil)
            {
                return;
            }
            var width = 310f;
            var rect = new Rect((Screen.width - width) * 0.5f, 22f, width, 36f);
            var style = new GUIStyle(GUI.skin.box)
            {
                alignment = TextAnchor.MiddleCenter,
                fontSize = 17,
                fontStyle = FontStyle.Bold
            };
            GUI.Box(rect, _toggleNotice, style);
        }

        private void ToggleLanguage()
        {
            var stopwatch = Stopwatch.StartNew();
            ChineseEnabled = !ChineseEnabled;
            if (ChineseEnabled)
            {
                ChineseFontScale = Mathf.Clamp(_fontScale.Value, 1f, 1.10f);
                ApplyCachedLanguageValues(true);
                RelocalizeVisibleTextComponents(true);
                RelocalizeVisibleTypewriters(true);
                RefreshVisibleWorldPopups();
                // IMGUI uses the game's legacy font rather than our TMP fallback. Keep this
                // notice ASCII-only so it remains readable even before a CJK font is available.
                _toggleNotice = "ZH-CN enabled (F9 to switch)";
            }
            else
            {
                RestoreOriginalValues();
                RelocalizeVisibleTextComponents(false);
                RelocalizeVisibleTypewriters(false);
                RefreshVisibleWorldPopups();
                _toggleNotice = "English enabled (F9 to switch)";
            }
            stopwatch.Stop();
            _toggleNoticeUntil = Time.realtimeSinceStartup + 2.5f;
            Log.LogInfo(
                $"游戏内语言已切换为：{(ChineseEnabled ? "简体中文" : "English")}；" +
                $"同步应用耗时 {stopwatch.Elapsed.TotalMilliseconds:F1} ms。");
        }

        private static void LoadCatalog()
        {
            Translations.Clear();
            LinkTargetTranslations.Clear();
            SourceHashCache.Clear();
            KnownLocalizationPairs.Clear();
            KnownSourcesByTranslation.Clear();
            NormalizedSourceTargets.Clear();
            AmbiguousNormalizedSources.Clear();
            ForwardCompositePairs.Clear();
            ReverseCompositePairs.Clear();
            ShortReverseCompositePairs.Clear();
            ForwardCompositePrefixes.Clear();
            ReverseCompositePrefixes.Clear();
            ForwardCompositeCache.Clear();
            ReverseCompositeCache.Clear();
            TranslationPatterns.Clear();
            BufferLineForwardCache.Clear();
            BufferLineReverseCache.Clear();
            SpeakerNameForward.Clear();
            SpeakerNameReverse.Clear();
            CompositePairsDirty = true;
            KnownPairsVersion++;
            var path = Path.Combine(PluginDirectory, "catalog.zh-CN.json");
            if (!File.Exists(path))
            {
                Log.LogError($"未找到翻译目录：{path}");
                return;
            }

            Dictionary<string, string> loaded;
            try
            {
                loaded = JsonConvert.DeserializeObject<Dictionary<string, string>>(
                    File.ReadAllText(path, Encoding.UTF8));
            }
            catch (Exception exception)
            {
                Log.LogError($"翻译目录无法解析：{exception}");
                return;
            }
            if (loaded == null)
            {
                Log.LogError("翻译目录无法解析。");
                return;
            }

            var invalidTranslationKeys = 0;
            foreach (var pair in loaded)
            {
                if (IsSha256Fingerprint(pair.Key) && !string.IsNullOrEmpty(pair.Value))
                {
                    Translations[pair.Key] = pair.Value;
                }
                else if (!string.IsNullOrEmpty(pair.Key))
                {
                    invalidTranslationKeys++;
                }
            }
            foreach (var pair in RuntimeLocalizationAliases)
            {
                RegisterKnownLocalizationPair(pair.Key, pair.Value);
            }
            if (invalidTranslationKeys > 0)
            {
                Log.LogError(
                    $"翻译目录含 {invalidTranslationKeys} 个非 SHA-256 键；" +
                    "这通常表示目录与插件版本不匹配，相关条目已忽略。请重新构建补丁。");
            }

            var linkPath = Path.Combine(PluginDirectory, "link_targets.zh-CN.json");
            if (!File.Exists(linkPath))
            {
                Log.LogError($"未找到脚注链接目录：{linkPath}");
                return;
            }
            try
            {
                var links = JsonConvert.DeserializeObject<Dictionary<string, string>>(
                    File.ReadAllText(linkPath, Encoding.UTF8));
                if (links != null)
                {
                    var invalidLinkKeys = 0;
                    foreach (var pair in links)
                    {
                        if (IsSha256Fingerprint(pair.Key) && !string.IsNullOrEmpty(pair.Value))
                        {
                            LinkTargetTranslations[pair.Key] = pair.Value;
                        }
                        else if (!string.IsNullOrEmpty(pair.Key))
                        {
                            invalidLinkKeys++;
                        }
                    }
                    if (invalidLinkKeys > 0)
                    {
                        Log.LogError(
                            $"脚注链接目录含 {invalidLinkKeys} 个非 SHA-256 键；" +
                            "相关别名已忽略。请重新构建补丁。");
                    }
                }
            }
            catch (Exception exception)
            {
                Log.LogError($"脚注链接目录无法解析：{exception}");
            }

            var patternsPath = Path.Combine(PluginDirectory, "patterns.zh-CN.json");
            if (!File.Exists(patternsPath))
            {
                Log.LogWarning("未找到变量模式目录；[q=...] 替换文本将退回逐词处理。请重新构建补丁。");
                return;
            }
            try
            {
                var patternFile = JsonConvert.DeserializeObject<TranslationPatternFile>(
                    File.ReadAllText(patternsPath, Encoding.UTF8));
                if (patternFile?.patterns != null)
                {
                    foreach (var entry in patternFile.patterns)
                    {
                        if (entry?.segments == null || entry.tokens == null ||
                            string.IsNullOrEmpty(entry.translation) ||
                            entry.segments.Length != entry.tokens.Length + 1)
                        {
                            continue;
                        }
                        entry.TotalLiteralLength = entry.segments.Sum(segment => segment?.len ?? 0);
                        TranslationPatterns.Add(entry);
                    }
                    TranslationPatterns.Sort((left, right) =>
                        right.TotalLiteralLength.CompareTo(left.TotalLiteralLength));
                }
            }
            catch (Exception exception)
            {
                Log.LogError($"变量模式目录无法解析：{exception}");
                TranslationPatterns.Clear();
            }
        }

        private static bool IsSha256Fingerprint(string value)
        {
            if (value == null || value.Length != 64)
            {
                return false;
            }
            foreach (var character in value)
            {
                if (!((character >= '0' && character <= '9') ||
                      (character >= 'a' && character <= 'f') ||
                      (character >= 'A' && character <= 'F')))
                {
                    return false;
                }
            }
            return true;
        }

        private static void InstallChineseFont()
        {
            try
            {
                var fontPath = Path.Combine(PluginDirectory, "font", "NotoSansCJKsc-Regular.otf");
                if (!File.Exists(fontPath))
                {
                    Log.LogError($"未找到中文字体：{fontPath}");
                    return;
                }

                ChineseFont = TMP_FontAsset.CreateFontAsset(
                    fontPath,
                    0,
                    48,
                    5,
                    GlyphRenderMode.SDFAA,
                    2048,
                    2048);
                if (ChineseFont == null)
                {
                    Log.LogError("TextMeshPro 无法创建中文字体资产。");
                    return;
                }

                ChineseFont.name = "Noto Sans CJK SC Regular Dynamic (TravellingCN)";
                ChineseFont.atlasPopulationMode = AtlasPopulationMode.Dynamic;
                ChineseFont.isMultiAtlasTexturesEnabled = true;
                UnityEngine.Object.DontDestroyOnLoad(ChineseFont);

                if (!EnsureGlobalChineseFallback())
                {
                    Log.LogWarning("TMP 全局设置尚未就绪；稍后将再次挂载中文 fallback 字体。");
                }
                Log.LogInfo("中文动态字体已安装。字符将在首次显示时按需生成。");
            }
            catch (Exception exception)
            {
                Log.LogError($"安装中文字体失败：{exception}");
            }
        }

        private static bool EnsureGlobalChineseFallback()
        {
            if (ChineseFont == null)
            {
                return false;
            }
            if (GlobalFallbackInstalled)
            {
                return true;
            }
            try
            {
                var fallbacks = TMP_Settings.fallbackFontAssets;
                if (fallbacks == null)
                {
                    fallbacks = new List<TMP_FontAsset>();
                    TMP_Settings.fallbackFontAssets = fallbacks;
                }
                if (!fallbacks.Contains(ChineseFont))
                {
                    fallbacks.Insert(0, ChineseFont);
                }
                GlobalFallbackInstalled = true;
                return true;
            }
            catch
            {
                return false;
            }
        }

        private static bool TryTranslate(string source, out string target)
        {
            target = null;
            if (!ChineseEnabled || string.IsNullOrEmpty(source))
            {
                return false;
            }
            return TryLookupTranslation(source, out target);
        }

        private static bool TryLookupTranslation(string source, out string target)
        {
            target = null;
            if (string.IsNullOrEmpty(source))
            {
                return false;
            }
            if (KnownLocalizationPairs.TryGetValue(source, out target))
            {
                return true;
            }
            if (!Translations.TryGetValue(GetSourceHash(source), out target))
            {
                // Dialogue System turns [[Footnote]] into TMP <link> markup
                // before assigning some lazy tooltip texts.  The catalog is
                // deliberately keyed by the original authored string, so
                // reconstruct that form from stable link IDs and then render
                // the translated links back to TMP markup.
                var canonical = CanonicalizeRenderedLinks(source);
                if (string.Equals(canonical, source, StringComparison.Ordinal) ||
                    !Translations.TryGetValue(GetSourceHash(canonical), out target))
                {
                    var normalized = NormalizeLookupText(source);
                    if (string.IsNullOrEmpty(normalized) ||
                        AmbiguousNormalizedSources.Contains(normalized) ||
                        !NormalizedSourceTargets.TryGetValue(normalized, out target))
                    {
                        if (!TrySpliceDecoratedTranslation(source, out target))
                        {
                            target = null;
                            return false;
                        }
                    }
                }
                else
                {
                    target = RenderWikiLinksForTmp(target, canonical);
                }
            }
            if (string.Equals(source, target, StringComparison.Ordinal))
            {
                target = null;
                return false;
            }
            RegisterKnownLocalizationPair(source, target);
            return true;
        }

        private static void RegisterKnownLocalizationPair(string source, string target)
        {
            if (string.IsNullOrEmpty(source) || string.IsNullOrEmpty(target))
            {
                return;
            }
            if (KnownLocalizationPairs.TryGetValue(source, out var oldTarget))
            {
                if (string.Equals(oldTarget, target, StringComparison.Ordinal))
                {
                    return;
                }
                if (KnownSourcesByTranslation.TryGetValue(oldTarget, out var oldSources))
                {
                    oldSources.Remove(source);
                    if (oldSources.Count == 0)
                    {
                        KnownSourcesByTranslation.Remove(oldTarget);
                    }
                }
            }
            KnownLocalizationPairs[source] = target;
            if (!KnownSourcesByTranslation.TryGetValue(target, out var sources))
            {
                sources = new HashSet<string>(StringComparer.Ordinal);
                KnownSourcesByTranslation[target] = sources;
            }
            sources.Add(source);
            RegisterNormalizedLocalizationPair(source, target);
            // 只标记脏；版本号与重建在 EnsureCompositePairs 限频推进，
            // 避免每条新字幕都失效全部复合缓存。
            CompositePairsDirty = true;
        }

        private static string NormalizeLookupText(string value)
        {
            if (string.IsNullOrEmpty(value))
            {
                return string.Empty;
            }
            var normalized = WikiLinkRegex.Replace(value, match => match.Groups["id"].Value);
            normalized = CosmeticMarkupRegex.Replace(normalized, string.Empty);
            normalized = normalized
                .Replace("\r\n", "\n")
                .Replace('\r', '\n')
                .Replace('\u00a0', ' ')
                .Trim();
            return normalized;
        }

        private static void RegisterNormalizedLocalizationPair(string source, string target)
        {
            var normalizedSource = NormalizeLookupText(source);
            if (string.IsNullOrEmpty(normalizedSource) ||
                AmbiguousNormalizedSources.Contains(normalizedSource))
            {
                return;
            }
            if (!NormalizedSourceTargets.TryGetValue(normalizedSource, out var existingTarget))
            {
                NormalizedSourceTargets[normalizedSource] = target;
                return;
            }

            // Mark a visible form ambiguous only when it would genuinely render
            // different words. Cosmetic authored tags may legitimately differ.
            if (!string.Equals(
                    NormalizeLookupText(existingTarget),
                    NormalizeLookupText(target),
                    StringComparison.Ordinal))
            {
                NormalizedSourceTargets.Remove(normalizedSource);
                AmbiguousNormalizedSources.Add(normalizedSource);
            }
        }

        private static bool TryGetUniqueSource(string translated, out string source)
        {
            source = null;
            if (string.IsNullOrEmpty(translated) ||
                !KnownSourcesByTranslation.TryGetValue(translated, out var sources) ||
                sources.Count != 1)
            {
                return false;
            }
            source = sources.First();
            return true;
        }

        private static void EnsureCompositePairs()
        {
            if (!CompositePairsDirty)
            {
                return;
            }
            // 对话进行中每条新字幕都会登记新对；全量重建（排序+去装饰归组）
            // 是毫秒到几十毫秒级的开销，限频到每 2 秒一次，避免周期性卡顿。
            var tick = Environment.TickCount;
            if (ForwardCompositePairs.Count > 0 &&
                tick - LastCompositeRebuildTick >= 0 &&
                tick - LastCompositeRebuildTick < 2000)
            {
                return;
            }
            LastCompositeRebuildTick = tick;
            ForwardCompositePairs = KnownLocalizationPairs
                .Where(pair => !string.IsNullOrEmpty(pair.Key) && pair.Key.Length >= 4)
                .Select(pair => new KeyValuePair<string, string>(
                    CanonicalizeCosmeticMarkup(pair.Key),
                    pair.Value))
                .OrderByDescending(pair => pair.Key.Length)
                .ToList();
            ReverseCompositePairs = KnownSourcesByTranslation
                .Where(pair => !string.IsNullOrEmpty(pair.Key) &&
                    pair.Key.Length >= 4 && pair.Value.Count == 1)
                .Select(pair => new KeyValuePair<string, string>(
                    CanonicalizeCosmeticMarkup(pair.Key),
                    pair.Value.First()))
                .OrderByDescending(pair => pair.Key.Length)
                .ToList();
            ShortReverseCompositePairs = KnownSourcesByTranslation
                .Where(pair => !string.IsNullOrEmpty(pair.Key) &&
                    pair.Key.Length < 4 && pair.Value.Count == 1 && ContainsCjk(pair.Key))
                .Select(pair => new KeyValuePair<string, string>(pair.Key, pair.Value.First()))
                .OrderByDescending(pair => pair.Key.Length)
                .ToList();
            AppendStrippedCompositePairs(ForwardCompositePairs);
            AppendStrippedReverseCompositePairs();
            ForwardCompositePrefixes = new HashSet<string>(
                ForwardCompositePairs.Select(pair => pair.Key.Substring(0, 4)),
                StringComparer.Ordinal);
            ReverseCompositePrefixes = new HashSet<string>(
                ReverseCompositePairs.Select(pair => pair.Key.Substring(0, 4)),
                StringComparer.Ordinal);
            KnownPairsVersion++;
            CompositePairsDirty = false;
        }

        private static void AppendStrippedCompositePairs(List<KeyValuePair<string, string>> pairs)
        {
            // 累计对话缓冲里同一行的新/旧版本只有颜色透明度等装饰差异
            // （#231a17ff 亮、#231a17cc 暗）。登记去掉装饰的句级变体，让两种
            // 形态都能整行替换；过短键的子串替换误伤风险高，不生成变体。
            var seen = new HashSet<string>(
                pairs.Select(pair => pair.Key),
                StringComparer.Ordinal);
            var additions = new List<KeyValuePair<string, string>>();
            foreach (var pair in pairs)
            {
                var strippedKey = StripDisplayDecorations(pair.Key);
                if (strippedKey.Length < 8 || !seen.Add(strippedKey))
                {
                    continue;
                }
                additions.Add(new KeyValuePair<string, string>(
                    strippedKey,
                    StripDisplayDecorations(pair.Value)));
            }
            if (additions.Count > 0)
            {
                pairs.AddRange(additions);
                pairs.Sort((left, right) => right.Key.Length.CompareTo(left.Key.Length));
            }
        }

        private static void AppendStrippedReverseCompositePairs()
        {
            // 反向（切回英文）不能只看 ReverseCompositePairs：同一译文可能从亮色、
            // 暗色等多个装饰源登记，唯一源过滤会把它们整体排除。改为按去装饰形态
            // 归组——若所有源去装饰后收敛为同一文本，则反向目标无歧义，可以登记。
            var sourcesByStrippedTarget = new Dictionary<string, HashSet<string>>(
                StringComparer.Ordinal);
            var strippedSourceByTarget = new Dictionary<string, string>(
                StringComparer.Ordinal);
            var ambiguous = new HashSet<string>(StringComparer.Ordinal);
            foreach (var pair in KnownSourcesByTranslation)
            {
                var strippedTarget = StripDisplayDecorations(pair.Key);
                if (strippedTarget.Length < 8 || ambiguous.Contains(strippedTarget))
                {
                    continue;
                }
                if (!sourcesByStrippedTarget.TryGetValue(strippedTarget, out var exactForms))
                {
                    exactForms = new HashSet<string>(StringComparer.Ordinal);
                    sourcesByStrippedTarget[strippedTarget] = exactForms;
                }
                exactForms.Add(pair.Key);
                foreach (var source in pair.Value)
                {
                    var strippedSource = StripDisplayDecorations(source);
                    if (strippedSourceByTarget.TryGetValue(strippedTarget, out var existing) &&
                        !string.Equals(existing, strippedSource, StringComparison.Ordinal))
                    {
                        ambiguous.Add(strippedTarget);
                        sourcesByStrippedTarget.Remove(strippedTarget);
                        strippedSourceByTarget.Remove(strippedTarget);
                        break;
                    }
                    strippedSourceByTarget[strippedTarget] = strippedSource;
                }
            }
            if (sourcesByStrippedTarget.Count == 0)
            {
                return;
            }
            var seen = new HashSet<string>(
                ReverseCompositePairs.Select(pair => pair.Key),
                StringComparer.Ordinal);
            var additions = new List<KeyValuePair<string, string>>();
            foreach (var pair in sourcesByStrippedTarget)
            {
                if (ambiguous.Contains(pair.Key) || !seen.Add(pair.Key))
                {
                    continue;
                }
                additions.Add(new KeyValuePair<string, string>(
                    pair.Key,
                    strippedSourceByTarget[pair.Key]));
            }
            if (additions.Count > 0)
            {
                ReverseCompositePairs.AddRange(additions);
                ReverseCompositePairs.Sort((left, right) => right.Key.Length.CompareTo(left.Key.Length));
            }
        }

        private static string RelocalizeCompositeText(
            string value,
            bool toChinese,
            bool allowLineSplice = true)
        {
            if (string.IsNullOrEmpty(value) || KnownLocalizationPairs.Count == 0)
            {
                return value;
            }
            if (toChinese)
            {
                if (KnownLocalizationPairs.TryGetValue(value, out var exactTarget))
                {
                    return exactTarget;
                }
                if (KnownSourcesByTranslation.ContainsKey(value))
                {
                    return value;
                }
                if (value.IndexOf('\n') < 0)
                {
                    if (TrySpliceDecoratedTranslation(value, out var splicedTarget) &&
                        !string.Equals(value, splicedTarget, StringComparison.Ordinal))
                    {
                        // 世界弹窗等边界只经过复合替换，不经过整串查找；装饰包裹或
                        // 变量替换的整句在这里直接命中并登记，供累计缓冲复用。
                        RegisterKnownLocalizationPair(value, splicedTarget);
                        return splicedTarget;
                    }
                }
                else if (allowLineSplice)
                {
                    // 累计对话缓冲一次包含多行；能力变动消息等行内带有逐词链接
                    // 与变量替换，整缓冲不可能命中。逐行拼接，让缓冲里的每一行
                    // 都能享受与单行边界相同的鲁棒查找。打字机中途重放时必须
                    // 关闭逐行拼接：行内容变化会让可见字符索引失配，正逐字显示
                    // 的行会整段提前闪现。
                    var lines = value.Split('\n');
                    var spliced = false;
                    for (var index = 0; index < lines.Length; index++)
                    {
                        var line = lines[index];
                        if (line.Length < 12 || !AsciiLetterRegex.IsMatch(line))
                        {
                            continue;
                        }
                        if (TrySpliceDecoratedTranslation(line, out var splicedLine) &&
                            !string.Equals(line, splicedLine, StringComparison.Ordinal))
                        {
                            lines[index] = splicedLine;
                            RegisterKnownLocalizationPair(line, splicedLine);
                            spliced = true;
                        }
                    }
                    if (spliced)
                    {
                        // 不提前返回：缓冲里可能还有只能靠复合对处理的行
                        // （含脚注链接的译文等），继续走下面的复合替换。
                        value = string.Join("\n", lines);
                    }
                }
            }
            else
            {
                if (TryGetUniqueSource(value, out var exactSource))
                {
                    return exactSource;
                }
                if (KnownLocalizationPairs.ContainsKey(value))
                {
                    return value;
                }
            }
            EnsureCompositePairs();
            var cache = toChinese ? ForwardCompositeCache : ReverseCompositeCache;
            if (cache.TryGetValue(value, out var cached) && cached.Version == KnownPairsVersion)
            {
                return cached.Result;
            }
            var candidates = toChinese ? ForwardCompositePairs : ReverseCompositePairs;
            // Protect TMP tags and their stable link/sprite identifiers. Only
            // visible text outside <...> participates in composite replacement.
            var protectedTags = new List<string>();
            var tagNonce = "\uE002TANTAG" + Time.frameCount.ToString("X") + "_";
            var result = CanonicalizeCosmeticMarkup(value);
            result = TmpTagRegex.Replace(result, match =>
            {
                var marker = tagNonce + protectedTags.Count + "\uE003";
                protectedTags.Add(match.Value);
                return marker;
            });
            var prefixes = toChinese ? ForwardCompositePrefixes : ReverseCompositePrefixes;
            var possibleMatch = false;
            for (var index = 0; index <= result.Length - 4; index++)
            {
                if (prefixes.Contains(result.Substring(index, 4)))
                {
                    possibleMatch = true;
                    break;
                }
            }
            if (!possibleMatch)
            {
                if (cache.Count >= 8192)
                {
                    cache.Clear();
                }
                cache[value] = new CompositeCacheEntry(KnownPairsVersion, value);
                return value;
            }
            var replacements = new List<KeyValuePair<string, string>>();
            var nonce = "\uE000TANCN" + Time.frameCount.ToString("X") + "_";
            foreach (var pair in candidates)
            {
                if (result.IndexOf(pair.Key, StringComparison.Ordinal) < 0)
                {
                    continue;
                }
                var marker = nonce + replacements.Count + "\uE001";
                var replaced = ReplaceRespectingAsciiWordBoundaries(result, pair.Key, marker);
                if (string.Equals(replaced, result, StringComparison.Ordinal))
                {
                    continue;
                }
                result = replaced;
                replacements.Add(new KeyValuePair<string, string>(marker, pair.Value));
            }
            foreach (var replacement in replacements)
            {
                result = result.Replace(replacement.Key, replacement.Value);
            }
            if (toChinese &&
                Translations.TryGetValue(GetSourceHash("Me"), out var translatedMe))
            {
                // "Me" is only two characters, so it is deliberately absent
                // from the ordinary >=4-character composite table.  Dialogue
                // history embeds it as a speaker token inside a larger rich-
                // text buffer; replace only the standalone visible token.
                result = StandaloneMeRegex.Replace(result, translatedMe);
            }
            for (var index = 0; index < protectedTags.Count; index++)
            {
                result = result.Replace(tagNonce + index + "\uE003", protectedTags[index]);
            }
            result = RestoreCanonicalCosmeticMarkup(result);
            if (!toChinese && !ChineseEnabled && ContainsCjk(result))
            {
                // F9 切英文后仍含中文：反向恢复未能覆盖的串，诊断采集用。
                DiagRecord("Reverse", "stuck", value);
            }
            if (cache.Count >= 8192)
            {
                cache.Clear();
            }
            cache[value] = new CompositeCacheEntry(KnownPairsVersion, result);
            return result;
        }

        private static bool IsAsciiLetter(char value)
        {
            return (value >= 'A' && value <= 'Z') || (value >= 'a' && value <= 'z');
        }

        private static string ReplaceRespectingAsciiWordBoundaries(
            string text,
            string key,
            string marker)
        {
            // 复合对按子串替换；键的字母边缘若不要求词界，"Trace" 会劈开
            // "Traces" 产出"痕迹s"这类半译僵尸串。仅当键的首/尾是 ASCII
            // 字母时，要求匹配位置的相邻字符不是 ASCII 字母。
            var guardLeft = IsAsciiLetter(key[0]);
            var guardRight = IsAsciiLetter(key[key.Length - 1]);
            if (!guardLeft && !guardRight)
            {
                return text.Replace(key, marker);
            }
            StringBuilder builder = null;
            var searchFrom = 0;
            while (searchFrom <= text.Length - key.Length)
            {
                var at = text.IndexOf(key, searchFrom, StringComparison.Ordinal);
                if (at < 0)
                {
                    break;
                }
                var leftOk = !guardLeft || at == 0 || !IsAsciiLetter(text[at - 1]);
                var rightIndex = at + key.Length;
                var rightOk = !guardRight ||
                    rightIndex >= text.Length ||
                    !IsAsciiLetter(text[rightIndex]);
                if (!leftOk || !rightOk)
                {
                    searchFrom = at + 1;
                    continue;
                }
                if (builder == null)
                {
                    builder = new StringBuilder(text.Length);
                }
                builder.Append(text, searchFrom, at - searchFrom);
                builder.Append(marker);
                searchFrom = rightIndex;
            }
            if (builder == null)
            {
                return text;
            }
            builder.Append(text, searchFrom, text.Length - searchFrom);
            return builder.ToString();
        }

        private static string CanonicalizeCosmeticMarkup(string value)
        {
            if (string.IsNullOrEmpty(value))
            {
                return value;
            }
            return CosmeticMarkupRegex.Replace(value, match =>
                "\uE100" +
                (match.Groups["close"].Success ? "/" : string.Empty) +
                match.Groups["name"].Value.ToLowerInvariant() +
                "\uE101");
        }

        private static string RestoreCanonicalCosmeticMarkup(string value)
        {
            if (string.IsNullOrEmpty(value))
            {
                return value;
            }
            return CanonicalCosmeticMarkupRegex.Replace(value, match =>
                "<" +
                (match.Groups["close"].Success ? "/" : string.Empty) +
                match.Groups["name"].Value.ToLowerInvariant() +
                ">");
        }

        private static string StripDisplayDecorations(string value)
        {
            if (string.IsNullOrEmpty(value))
            {
                return string.Empty;
            }
            var stripped = TmpTagRegex.Replace(value, string.Empty);
            if (stripped.IndexOfAny(InvisibleFormatChars) >= 0)
            {
                foreach (var invisible in InvisibleFormatChars)
                {
                    stripped = stripped.Replace(invisible.ToString(), string.Empty);
                }
            }
            return stripped;
        }

        private static bool TryFindPatternSegment(
            string value,
            TranslationPatternSegment segment,
            int fromIndex,
            bool isLastSegment,
            out int start)
        {
            start = -1;
            if (segment == null || segment.len < 0)
            {
                return false;
            }
            if (segment.len == 0)
            {
                // 变量位于句首/句尾时没有该侧字面锚点；尾段空锚定到串尾。
                start = isLastSegment ? value.Length : fromIndex;
                return true;
            }
            if (segment.len > value.Length - fromIndex ||
                string.IsNullOrEmpty(segment.first) ||
                string.IsNullOrEmpty(segment.last))
            {
                return false;
            }
            var first = segment.first[0];
            var last = segment.last[0];
            for (var index = fromIndex; index <= value.Length - segment.len; index++)
            {
                if (value[index] != first || value[index + segment.len - 1] != last)
                {
                    continue;
                }
                if (string.Equals(
                        GetSourceHash(value.Substring(index, segment.len)),
                        segment.sha,
                        StringComparison.Ordinal))
                {
                    start = index;
                    return true;
                }
            }
            return false;
        }

        private static bool TryAlignTranslationPattern(
            TranslationPatternEntry pattern,
            string visible,
            out string[] captures)
        {
            captures = new string[pattern.tokens.Length];
            var position = 0;
            for (var index = 0; index < pattern.segments.Length; index++)
            {
                var isLast = index == pattern.segments.Length - 1;
                if (!TryFindPatternSegment(
                        visible, pattern.segments[index], position, isLast, out var start))
                {
                    return false;
                }
                if (index == 0 && pattern.segments[0].len > 0 && start != 0)
                {
                    // 整串锚定：非空首段必须从串头开始，避免命中子串误译。
                    return false;
                }
                if (index > 0)
                {
                    captures[index - 1] = visible.Substring(position, start - position);
                }
                position = start + pattern.segments[index].len;
            }
            return position == visible.Length;
        }

        private static string TranslateDynamicValue(string value, int depth)
        {
            // 变量捕获值本身可能是复合文本：技能/场所名单（"A in a B；或 C"）、
            // 性相列表（"Influence, Winter"）等。按 精确 → 子句拆分 → 模式 →
            // 连接词拆分 → 复合词级 的顺序尽力本地化。
            if (string.IsNullOrEmpty(value))
            {
                return value;
            }
            if (Translations.TryGetValue(GetSourceHash(value), out var exact))
            {
                return exact;
            }
            if (depth >= 8)
            {
                return RelocalizeCompositeText(value, true);
            }
            var trimmedValue = value.Trim();
            // 牛津逗号切子句边界最优先；"{0} in a {1}" 等模式先于普通
            // " or " 拆分，否则连接词会劈开模式宾语。
            foreach (var separator in DynamicClauseSeparators)
            {
                if (TrySplitTranslateDynamicValue(trimmedValue, separator, "；或 ", depth, out var claused))
                {
                    return claused;
                }
            }
            if (TryMatchTranslationPattern(trimmedValue, depth + 1, 0, out var patterned))
            {
                return patterned;
            }
            foreach (var (separator, joiner) in DynamicListJoins)
            {
                if (TrySplitTranslateDynamicValue(trimmedValue, separator, joiner, depth, out var listed))
                {
                    return listed;
                }
            }
            return RelocalizeCompositeText(value, true);
        }

        private static bool TrySplitTranslateDynamicValue(
            string value,
            string separator,
            string joiner,
            int depth,
            out string translated)
        {
            translated = null;
            if (value.IndexOf(separator, StringComparison.Ordinal) < 0)
            {
                return false;
            }
            var parts = value
                .Split(new[] { separator }, StringSplitOptions.None)
                .Select(part => part.Trim())
                .ToArray();
            if (parts.Length < 2 || parts.Any(string.IsNullOrEmpty))
            {
                return false;
            }
            var translatedParts = parts
                .Select(part => TranslateDynamicValue(part, depth + 1))
                .ToArray();
            if (translatedParts.SequenceEqual(parts))
            {
                return false;
            }
            translated = string.Join(joiner, translatedParts);
            return true;
        }

        private static string SubstituteTranslationPattern(
            TranslationPatternEntry pattern,
            string[] captures,
            int depth)
        {
            var result = pattern.translation;
            for (var index = 0; index < pattern.tokens.Length; index++)
            {
                var value = TranslateDynamicValue(captures[index] ?? string.Empty, depth + 1);
                result = result.Replace(pattern.tokens[index], value);
            }
            return result;
        }

        private static bool TryMatchTranslationPattern(
            string visible,
            int depth,
            int minLiteralLength,
            out string translated)
        {
            translated = null;
            if (string.IsNullOrEmpty(visible) || visible.Length > 2048)
            {
                return false;
            }
            foreach (var pattern in TranslationPatterns)
            {
                // 锚点过弱的模式（如 "[q=alias.formal.fr]." 只有一个句号字面量）
                // 会吞掉任意以句点结尾的整句，把未译英文变成"英文+。"僵尸串。
                // 整句拼接路径要求足够的字面锚点；动态参数递归路径的捕获值
                // 已经过拆分，允许使用弱锚点模式（如 "{0} in a {1}"）。
                if (pattern.TotalLiteralLength < minLiteralLength ||
                    pattern.TotalLiteralLength > visible.Length)
                {
                    continue;
                }
                if (!TryAlignTranslationPattern(pattern, visible, out var captures))
                {
                    continue;
                }
                // 捕获值是别名、数字等动态内容；含装饰/控制括号说明首段之前的
                // 编号、概率前缀等装饰被误吞进变量槽，该匹配必须拒绝。
                if (captures.Any(capture =>
                        capture != null &&
                        (capture.IndexOf('<') >= 0 || capture.IndexOf('>') >= 0 ||
                         capture.IndexOf('[') >= 0 || capture.IndexOf(']') >= 0)))
                {
                    continue;
                }
                translated = SubstituteTranslationPattern(pattern, captures, depth);
                return true;
            }
            return false;
        }

        private static bool TrySpliceCandidate(
            string source,
            string candidate,
            string visible,
            bool allowWikiLinks,
            out string target)
        {
            target = null;
            // 过短或无字母的核心容易撞中目录里的标点/符号条目（如 "."→"。"），
            // 把隐形的版式分隔符也"翻译"掉；拼接只服务句级文本。
            if (candidate.Length < 2 || !AsciiLetterRegex.IsMatch(candidate))
            {
                return false;
            }
            string coreTranslation;
            if (!Translations.TryGetValue(GetSourceHash(candidate), out coreTranslation) &&
                !TryMatchTranslationPattern(candidate, 0, 8, out coreTranslation))
            {
                return false;
            }
            if (string.IsNullOrEmpty(coreTranslation) ||
                (!allowWikiLinks &&
                    coreTranslation.IndexOf("[[", StringComparison.Ordinal) >= 0))
            {
                // 含脚注链接的译文必须走 authored 链接恢复路径，不能裸文本回填；
                // 调用方已完成链接规范化（allowWikiLinks）时除外。
                return false;
            }
            var at = source.IndexOf(candidate, StringComparison.Ordinal);
            if (at >= 0)
            {
                target = source.Substring(0, at) + coreTranslation +
                    source.Substring(at + candidate.Length);
                return true;
            }
            if (string.Equals(candidate, visible, StringComparison.Ordinal))
            {
                // 标签穿插在可视文本内部（如逐词链接），无法原位拼接；整串替换
                // 为纯译文，丢弃游戏附加的装饰样式，保证玩家看到完整中文。
                target = coreTranslation;
                return true;
            }
            return false;
        }

        private static bool TrySpliceDecoratedTranslation(string source, out string target)
        {
            // 游戏在显示前会装饰字符串（整体颜色包裹、选项编号、[97%] 检定概率、
            // 精灵图标），并先替换 [q=...] 变量。剥掉装饰后对可视核心做精确指纹
            // 或变量模式匹配，再把译文拼回原始装饰串。
            target = null;
            if (string.IsNullOrEmpty(source))
            {
                return false;
            }
            // 目录按 authored 原文（含 [[...]] 脚注）取指纹；游戏渲染后链接变成
            // <link> 标记。先把可识别的 <link> 还原为 authored 形态，含链接的整句
            // 才能命中指纹；命中后再把译文里的链接标签译出并渲染回 TMP 标记。
            var working = CanonicalizeRenderedLinks(source);
            var canonicalized = !string.Equals(working, source, StringComparison.Ordinal);
            var visible = StripDisplayDecorations(working).Trim();
            if (visible.Length == 0)
            {
                return false;
            }
            var candidate = visible;
            for (var step = 0; step < 4 && candidate.Length > 0; step++)
            {
                if (TrySpliceCandidate(working, candidate, visible, canonicalized, out var spliced))
                {
                    target = spliced;
                    break;
                }
                // 修剪顺序：编号 → [97%] → [[wiki]] 提示标签 → 普通 [标签]。
                // 方括号规则排除了 '['，不会从 [[...]] 中间截断；wiki 标签在
                // 编号/概率之后剥离，兼容 "2. [97%] [[[label]]] 正文" 形态。
                var trimmed = LeadingBracketTagRegex.Replace(
                    LeadingWikiLabelRegex.Replace(
                        LeadingChanceRegex.Replace(
                            LeadingNumberingRegex.Replace(candidate, string.Empty),
                            string.Empty),
                        string.Empty),
                    string.Empty).Trim();
                if (string.Equals(trimmed, candidate, StringComparison.Ordinal))
                {
                    break;
                }
                candidate = trimmed;
            }
            if (target == null)
            {
                return false;
            }
            if (canonicalized)
            {
                var wikiTarget = LocalizeWikiLinkLabels(target);
                var rendered = RenderWikiLinksForTmp(wikiTarget, working);
                if (rendered.IndexOf("[[", StringComparison.Ordinal) >= 0)
                {
                    // 译文链接数与 authored 形态对不上时放弃链接保真，
                    // 至少让玩家看到完整中文而不是原样英文。
                    target = WikiLinkRegex.Replace(
                        wikiTarget,
                        match => match.Groups["id"].Value);
                }
                else
                {
                    target = RestoreAuthoredLinkStyles(rendered, source, wikiTarget);
                }
            }
            else
            {
                if (target.IndexOf("[[", StringComparison.Ordinal) >= 0)
                {
                    // 非规范化路径不得把 authored [[...]] 残片带上屏：
                    // 含脚注的整句必须走 RichLink 边界的 authored 恢复路径。
                    target = null;
                    return false;
                }
                if (!string.Equals(candidate, visible, StringComparison.Ordinal))
                {
                    target = TranslateSplicedLabelPrefix(target, visible, candidate);
                }
            }
            return true;
        }

        private static string TranslateSplicedLabelPrefix(
            string target,
            string visible,
            string candidate)
        {
            // 选项提示标签（如 [Nina in Antibes?]）随编号/[97%] 前缀一起被修剪；
            // 正文命中后把标签本体单独查目录译出，替换它在拼接结果里的最后一次
            // 出现（链接 id 属性先于可见标签出现，末次出现才是可见文本）。
            var at = visible.IndexOf(candidate, StringComparison.Ordinal);
            if (at <= 0)
            {
                return target;
            }
            var labelCore = visible
                .Substring(0, at)
                .Trim()
                .Trim('[', ']', ' ', '-', '–', '—');
            if (labelCore.Length < 2 || !AsciiLetterRegex.IsMatch(labelCore))
            {
                return target;
            }
            if (!TryLookupTranslation(labelCore, out var labelTranslation))
            {
                return target;
            }
            var labelAt = target.LastIndexOf(labelCore, StringComparison.Ordinal);
            if (labelAt < 0)
            {
                return target;
            }
            return target.Substring(0, labelAt) + labelTranslation +
                target.Substring(labelAt + labelCore.Length);
        }

        private static string RelocalizeDialogueBufferSource(string value)
        {
            // 逐行映射还原：缓冲里的中文行都来自本插件在词条/拼接边界登记的
            // 精确对（英文渲染形态 ↔ 中文渲染形态），逐行反查即可无损还原，
            // 不再依赖复合词级替换的往返猜测。
            return RestoreDialogueBufferEnglish(value);
        }

        private static string LocalizeDialogueBufferChinese(string buffer)
        {
            if (string.IsNullOrEmpty(buffer))
            {
                return buffer;
            }
            var lines = buffer.Split('\n');
            var changed = false;
            for (var index = 0; index < lines.Length; index++)
            {
                var localized = LocalizeBufferLineForward(lines[index]);
                if (!string.Equals(localized, lines[index], StringComparison.Ordinal))
                {
                    lines[index] = localized;
                    changed = true;
                }
            }
            return changed ? string.Join("\n", lines) : buffer;
        }

        private static string LocalizeBufferLineForward(string line)
        {
            if (line.Length < 2 || !AsciiLetterRegex.IsMatch(line))
            {
                // 已是中文的行、隐形分隔符等无需处理。
                return line;
            }
            if (BufferLineForwardCache.TryGetValue(line, out var cached))
            {
                return cached;
            }
            var result = line;
            if (TryLookupTranslation(line, out var whole))
            {
                result = whole;
            }
            else
            {
                var head = string.Empty;
                var rest = line;
                var speakerMatch = BufferSpeakerPrefixRegex.Match(line);
                if (speakerMatch.Success)
                {
                    head = speakerMatch.Groups["head"].Value;
                    rest = speakerMatch.Groups["rest"].Value;
                    var nameMatch = BufferSpeakerNameRegex.Match(head);
                    if (nameMatch.Success)
                    {
                        var name = nameMatch.Groups["name"].Value;
                        if (TryLookupTranslation(name, out var translatedName))
                        {
                            head = head.Replace(name, translatedName);
                            SpeakerNameForward[name] = translatedName;
                            SpeakerNameReverse[translatedName] = name;
                        }
                    }
                }
                var colorMatch = BufferOuterColorRegex.Match(rest);
                if (colorMatch.Success)
                {
                    var content = colorMatch.Groups["content"].Value;
                    if (TryLookupTranslation(content, out var translatedContent))
                    {
                        rest = colorMatch.Groups["open"].Value +
                            translatedContent +
                            colorMatch.Groups["close"].Value;
                    }
                    else if (TrySpliceDecoratedTranslation(content, out var splicedContent) &&
                        !string.Equals(content, splicedContent, StringComparison.Ordinal))
                    {
                        // 行内还有 sprite/链接/变量等装饰或 [q=] 已替换的整句：
                        // 精确指纹命不中，走与单行边界相同的装饰拼接兜底。
                        rest = colorMatch.Groups["open"].Value +
                            splicedContent +
                            colorMatch.Groups["close"].Value;
                        RegisterKnownLocalizationPair(content, splicedContent);
                    }
                    else if (!ContainsCjk(content))
                    {
                        DiagRecord("BufferLine", "miss", content);
                    }
                }
                else if (TrySpliceDecoratedTranslation(rest, out var splicedRest) &&
                    !string.Equals(rest, splicedRest, StringComparison.Ordinal))
                {
                    rest = splicedRest;
                    RegisterKnownLocalizationPair(rest, splicedRest);
                }
                else if (!speakerMatch.Success && !ContainsCjk(rest))
                {
                    DiagRecord("BufferLine", "miss", rest);
                }
                result = head + rest;
            }
            BufferLineForwardCache[line] = result;
            if (!string.Equals(result, line, StringComparison.Ordinal))
            {
                BufferLineReverseCache[result] = line;
            }
            return result;
        }

        private static string RestoreDialogueBufferEnglish(string buffer)
        {
            if (string.IsNullOrEmpty(buffer) || !ContainsCjk(buffer))
            {
                return buffer;
            }
            var lines = buffer.Split('\n');
            var changed = false;
            for (var index = 0; index < lines.Length; index++)
            {
                var restored = RestoreBufferLineEnglish(lines[index]);
                if (!string.Equals(restored, lines[index], StringComparison.Ordinal))
                {
                    lines[index] = restored;
                    changed = true;
                }
            }
            return changed ? string.Join("\n", lines) : buffer;
        }

        private static bool TryGetAnyKnownSource(string translated, out string source)
        {
            // 唯一源反查失败时的宽容兜底：同一中文渲染形态常由亮色/暗色等多个
            // 装饰变体登记，Count!=1 就拒答会让 F9 整行卡中文。任取一个已知源
            // （优先含字母、最长）远好于不还原；下次游戏推送真实英文缓冲时
            // 缓存会自动校正。
            source = null;
            if (string.IsNullOrEmpty(translated) ||
                !KnownSourcesByTranslation.TryGetValue(translated, out var sources) ||
                sources.Count == 0)
            {
                return false;
            }
            source = sources
                .OrderByDescending(candidate => AsciiLetterRegex.IsMatch(candidate))
                .ThenByDescending(candidate => candidate.Length)
                .First();
            return true;
        }

        private static bool TryRestoreContentEnglish(string content, out string source)
        {
            // 内容级反向链：唯一源 → 宽容任取 → 反向复合对（覆盖"获得 X（N）"
            // 这类模式渲染行，它们从未登记过整行精确对）。
            if (TryGetUniqueSource(content, out source) ||
                TryGetAnyKnownSource(content, out source))
            {
                return true;
            }
            var reversed = RelocalizeCompositeText(content, false);
            if (!string.Equals(reversed, content, StringComparison.Ordinal) &&
                !ContainsCjk(reversed))
            {
                source = reversed;
                return true;
            }
            source = null;
            return false;
        }

        private static string RestoreBufferLineEnglish(string line)
        {
            if (!ContainsCjk(line))
            {
                return line;
            }
            if (BufferLineReverseCache.TryGetValue(line, out var cached))
            {
                return cached;
            }
            var result = line;
            if (TryGetUniqueSource(line, out var exactSource))
            {
                result = exactSource;
            }
            else if (TryGetAnyKnownSource(line, out var anySource))
            {
                result = anySource;
            }
            else
            {
                var head = string.Empty;
                var rest = line;
                var speakerMatch = BufferSpeakerPrefixRegex.Match(line);
                if (speakerMatch.Success)
                {
                    head = speakerMatch.Groups["head"].Value;
                    rest = speakerMatch.Groups["rest"].Value;
                    var nameMatch = BufferSpeakerNameRegex.Match(head);
                    if (nameMatch.Success)
                    {
                        var name = nameMatch.Groups["name"].Value;
                        if (SpeakerNameReverse.TryGetValue(name, out var englishName))
                        {
                            head = head.Replace(name, englishName);
                        }
                        else if (TryGetAnyKnownSource(name, out var knownName))
                        {
                            head = head.Replace(name, knownName);
                        }
                    }
                }
                var colorMatch = BufferOuterColorRegex.Match(rest);
                if (colorMatch.Success)
                {
                    var content = colorMatch.Groups["content"].Value;
                    if (TryRestoreContentEnglish(content, out var contentSource))
                    {
                        rest = colorMatch.Groups["open"].Value +
                            contentSource +
                            colorMatch.Groups["close"].Value;
                    }
                    else
                    {
                        DiagRecord("BufferLine", "stuck", content);
                    }
                }
                else if (TryRestoreContentEnglish(rest, out var restSource))
                {
                    rest = restSource;
                }
                else
                {
                    DiagRecord("BufferLine", "stuck", rest);
                }
                result = head + rest;
            }
            BufferLineReverseCache[line] = result;
            return result;
        }

        private static string ReplaceStandaloneVisibleToken(
            string value,
            string token,
            string replacement)
        {
            if (string.IsNullOrEmpty(value) || string.IsNullOrEmpty(token) ||
                value.IndexOf(token, StringComparison.Ordinal) < 0)
            {
                return value;
            }
            var result = new StringBuilder(value.Length);
            var searchFrom = 0;
            while (searchFrom < value.Length)
            {
                var index = value.IndexOf(token, searchFrom, StringComparison.Ordinal);
                if (index < 0)
                {
                    result.Append(value, searchFrom, value.Length - searchFrom);
                    break;
                }
                var end = index + token.Length;
                var leftBoundary = index == 0 || !char.IsLetterOrDigit(value[index - 1]);
                var rightBoundary = end == value.Length || !char.IsLetterOrDigit(value[end]);
                result.Append(value, searchFrom, index - searchFrom);
                if (leftBoundary && rightBoundary)
                {
                    result.Append(replacement);
                }
                else
                {
                    result.Append(token);
                }
                searchFrom = end;
            }
            return result.ToString();
        }

        private static void RelocalizeSubtitle(object subtitle, bool toChinese)
        {
            if (subtitle == null)
            {
                return;
            }
            var formatted = GetMemberValue(subtitle, "formattedText");
            if (formatted == null)
            {
                return;
            }
            var current = GetMemberValue(formatted, "text") as string;
            var source = current;
            if (OriginalMemberValues.TryGetValue(formatted, out var members) &&
                members.TryGetValue("text", out var savedSource))
            {
                source = savedSource;
            }
            else
            {
                if (toChinese || !ContainsCjk(current))
                {
                    GetOriginalMemberValue(formatted, "text", current);
                }
            }
            var replacement = toChinese
                ? RelocalizeCompositeText(source, true)
                : source;
            if (!string.Equals(current, replacement, StringComparison.Ordinal))
            {
                SetMemberValue(formatted, "text", replacement);
            }
            else if (toChinese && DiagLooksUntranslated(source))
            {
                DiagRecord("Subtitle", "miss", source);
            }
        }

        private static bool IsKnownTranslation(string source, string value)
        {
            return TryLookupTranslation(source, out var translated) &&
                string.Equals(value, translated, StringComparison.Ordinal);
        }

        private static string CanonicalizeRenderedLinks(string source)
        {
            if (string.IsNullOrEmpty(source) || source.IndexOf("<link", StringComparison.OrdinalIgnoreCase) < 0)
            {
                return source;
            }
            return RenderedLinkRegex.Replace(source, match =>
            {
                var linkId = match.Groups["id"].Value;
                if (string.IsNullOrEmpty(linkId) ||
                    !LinkTargetTranslations.ContainsKey(GetSourceHash(linkId)))
                {
                    return match.Value;
                }
                return "[[" + linkId + "]]";
            });
        }

        private static string RenderWikiLinksForTmp(string value, string canonicalSource)
        {
            if (string.IsNullOrEmpty(value) || value.IndexOf("[[", StringComparison.Ordinal) < 0)
            {
                return value;
            }
            var sourceLinks = WikiLinkRegex.Matches(canonicalSource ?? string.Empty);
            var targetLinks = WikiLinkRegex.Matches(value);
            if (sourceLinks.Count != targetLinks.Count)
            {
                return value;
            }
            var linkIndex = 0;
            return WikiLinkRegex.Replace(value, match =>
            {
                // The merged catalog stores reviewed Chinese link labels, but
                // the TMP link ID must remain the authored English lookup key.
                var linkId = sourceLinks[linkIndex++].Groups["id"].Value;
                var label = match.Groups["id"].Value;
                return "<link=\"" + linkId + "\">" + label + "</link>";
            });
        }

        private static string RestoreAuthoredLinkStyles(
            string localizedResult,
            string authoredResult,
            string localizedInput)
        {
            if (string.IsNullOrEmpty(localizedResult) ||
                string.IsNullOrEmpty(authoredResult) ||
                string.IsNullOrEmpty(localizedInput))
            {
                return localizedResult;
            }

            var localizedLinks = RenderedLinkRegex.Matches(localizedResult);
            var authoredLinks = RenderedLinkRegex.Matches(authoredResult);
            var localizedLabels = WikiLinkRegex.Matches(localizedInput);
            if (localizedLinks.Count == 0 ||
                localizedLinks.Count != authoredLinks.Count ||
                localizedLinks.Count != localizedLabels.Count)
            {
                return localizedResult;
            }

            var linkIndex = 0;
            return RenderedLinkRegex.Replace(localizedResult, match =>
            {
                var authoredMarkup = authoredLinks[linkIndex].Value;
                var localizedLabel = localizedLabels[linkIndex].Groups["id"].Value;
                linkIndex++;

                var labelReplaced = false;
                return RenderedLinkVisibleTextRegex.Replace(authoredMarkup, visibleMatch =>
                {
                    if (labelReplaced)
                    {
                        return visibleMatch.Value;
                    }
                    labelReplaced = true;
                    return localizedLabel;
                });
            });
        }

        private static string RestoreAuthoredWikiLinkTargets(
            string localizedInput,
            string authoredInput)
        {
            if (string.IsNullOrEmpty(localizedInput) ||
                string.IsNullOrEmpty(authoredInput))
            {
                return localizedInput;
            }

            var localizedLinks = WikiLinkRegex.Matches(localizedInput);
            var authoredLinks = WikiLinkRegex.Matches(authoredInput);
            if (localizedLinks.Count == 0 || localizedLinks.Count != authoredLinks.Count)
            {
                return localizedInput;
            }

            var linkIndex = 0;
            return WikiLinkRegex.Replace(localizedInput, _ =>
                "[[" + authoredLinks[linkIndex++].Groups["id"].Value + "]]"
            );
        }

        private static string LocalizeWikiLinkLabels(string input)
        {
            if (string.IsNullOrEmpty(input) || input.IndexOf("[[", StringComparison.Ordinal) < 0)
            {
                return input;
            }
            return WikiLinkRegex.Replace(input, match =>
            {
                var label = match.Groups["id"].Value;
                return TryLookupTranslation(label, out var translatedLabel)
                    ? "[[" + translatedLabel + "]]"
                    : match.Value;
            });
        }

        private static string LocalizeFormattedArgument(string argument)
        {
            if (string.IsNullOrEmpty(argument))
            {
                return argument;
            }
            if (TryLookupTranslation(argument, out var translatedArgument))
            {
                return translatedArgument;
            }

            // Dynamic Loc arguments can themselves be authored wiki tokens,
            // for example "[[Trace]]" in Skill.GetAdvice(). The catalog is
            // keyed by the bare label "Trace", so looking up the complete
            // bracketed argument can never succeed. Translate each visible
            // label now; the rich-link patches later restore the authored ID
            // before the game's resolver runs.
            var translatedLinks = LocalizeWikiLinkLabels(argument);
            return RelocalizeCompositeText(translatedLinks, true);
        }

        private static RichLinkLocalizationState PrepareRichLinkInput(ref string input)
        {
            if (!ChineseEnabled || string.IsNullOrEmpty(input))
            {
                return null;
            }

            // Translate the complete authored sentence before touching its
            // link labels. Localising only [[Trace]] / [[Worldly Skill]] leaves
            // the English sentence shell on screen even though the complete
            // description is present in the catalogue.
            var incomingInput = input;
            if (incomingInput.IndexOf("[q=", StringComparison.Ordinal) >= 0)
            {
                DiagRecord("RichLink", "arrive-q", incomingInput);
            }
            var authoredInput = RelocalizeCompositeText(incomingInput, false);
            var localizedInput = TryLookupTranslation(authoredInput, out var exactTranslation)
                ? exactTranslation
                : RelocalizeCompositeText(incomingInput, true);
            localizedInput = LocalizeWikiLinkLabels(localizedInput);
            var workingInput = RestoreAuthoredWikiLinkTargets(localizedInput, authoredInput);
            if (string.Equals(incomingInput, localizedInput, StringComparison.Ordinal) &&
                string.Equals(workingInput, localizedInput, StringComparison.Ordinal))
            {
                if (DiagLooksUntranslated(incomingInput))
                {
                    DiagRecord("RichLink", "miss", incomingInput);
                }
                return null;
            }
            if (authoredInput.IndexOf("[[", StringComparison.Ordinal) < 0 &&
                !string.Equals(authoredInput, localizedInput, StringComparison.Ordinal))
            {
                // 无链接的纯文本行在此登记 英文原文↔中文 精确对；缓冲逐行反查
                // （F9/英文源缓存）依赖这一对完整覆盖纯文本字幕。
                RegisterKnownLocalizationPair(authoredInput, localizedInput);
            }

            var state = new RichLinkLocalizationState(localizedInput, authoredInput);
            input = workingInput;
            return state;
        }

        private static void FinishRichLinkLocalization(
            RichLinkLocalizationState state,
            ref string result,
            string diagnostic)
        {
            if (state == null || string.IsNullOrEmpty(result))
            {
                return;
            }

            try
            {
                // This helper runs from a Harmony postfix. Reflectively
                // invoking the same method here re-enters this postfix and
                // recurses until the process dies with a stack overflow.
                // The first pass has already resolved the restored English
                // link ID, so use its output as the authored markup and only
                // replace the visible link label.
                var authoredResult = result;
                result = RestoreAuthoredLinkStyles(
                    authoredResult,
                    authoredResult,
                    state.LocalizedInput);
                if (!string.Equals(authoredResult, result, StringComparison.Ordinal))
                {
                    RegisterKnownLocalizationPair(authoredResult, result);
                }
            }
            catch (Exception exception)
            {
                Log.LogWarning($"{diagnostic}：{exception.GetBaseException().Message}");
            }
        }

        private static bool DiagLooksUntranslated(string value)
        {
            return DiagLoggingEnabled &&
                !string.IsNullOrEmpty(value) &&
                value.Length >= 6 &&
                !ContainsCjk(value) &&
                DiagEnglishWordRegex.IsMatch(value);
        }

        private static void DiagRecord(string boundary, string outcome, string raw)
        {
            if (!DiagLoggingEnabled || string.IsNullOrEmpty(DiagLogPath) || raw == null)
            {
                return;
            }
            var key = boundary + "\u001f" + outcome + "\u001f" + raw;
            lock (DiagLogLock)
            {
                if (!DiagSeenRecords.Add(key) || DiagLogCount >= 20000)
                {
                    return;
                }
                DiagLogCount++;
                try
                {
                    File.AppendAllText(
                        DiagLogPath,
                        "{\"b\":" + JsonConvert.SerializeObject(boundary) +
                        ",\"outcome\":" + JsonConvert.SerializeObject(outcome) +
                        ",\"raw\":" + JsonConvert.SerializeObject(raw) + "}\n",
                        Encoding.UTF8);
                }
                catch
                {
                    // Diagnostics must never disturb the game.
                }
            }
        }

        private static string GetSourceHash(string source)
        {
            if (SourceHashCache.TryGetValue(source, out var cached))
            {
                return cached;
            }
            byte[] digest;
            using (var sha256 = SHA256.Create())
            {
                digest = sha256.ComputeHash(Encoding.UTF8.GetBytes(source));
            }
            var builder = new StringBuilder(digest.Length * 2);
            foreach (var value in digest)
            {
                builder.Append(value.ToString("x2"));
            }
            var result = builder.ToString();
            if (SourceHashCache.Count >= 16384)
            {
                // Animated counters and clocks can generate unbounded unique
                // strings.  Keep the fingerprint cache useful but bounded.
                SourceHashCache.Clear();
            }
            SourceHashCache[source] = result;
            return result;
        }

        private static bool ContainsCjk(string value)
        {
            if (string.IsNullOrEmpty(value))
            {
                return false;
            }
            foreach (var character in value)
            {
                if ((character >= '\u3400' && character <= '\u9fff') ||
                    (character >= '\uf900' && character <= '\ufaff'))
                {
                    return true;
                }
            }
            return false;
        }

        private static bool CannotBeginLine(string token)
        {
            if (string.IsNullOrEmpty(token))
            {
                return false;
            }
            const string punctuation = "，。！？；：、）》】〕〉”’…—～·,.!?;:%)]}";
            return punctuation.IndexOf(token[0]) >= 0;
        }

        private static bool CannotEndLine(StringBuilder line)
        {
            if (line == null || line.Length == 0)
            {
                return false;
            }
            const string punctuation = "（《【〔〈“‘([{￥$";
            for (var index = line.Length - 1; index >= 0; index--)
            {
                if (!char.IsWhiteSpace(line[index]))
                {
                    return punctuation.IndexOf(line[index]) >= 0;
                }
            }
            return false;
        }

        private static string NextWrapToken(string text, ref int index, out bool visible)
        {
            visible = true;
            var start = index;
            if (text[index] == '<')
            {
                var end = text.IndexOf('>', index + 1);
                if (end >= 0)
                {
                    index = end + 1;
                    visible = false;
                    return text.Substring(start, index - start);
                }
            }
            if (text[index] == '[' && index + 1 < text.Length && text[index + 1] == '[')
            {
                var end = text.IndexOf("]]", index + 2, StringComparison.Ordinal);
                if (end >= 0)
                {
                    index = end + 2;
                    return text.Substring(start, index - start);
                }
            }
            if (text[index] == '\r' || text[index] == '\n')
            {
                if (text[index] == '\r' && index + 1 < text.Length && text[index + 1] == '\n')
                {
                    index += 2;
                    return "\r\n";
                }
                index++;
                return text.Substring(start, 1);
            }
            if (text[index] <= 0x7f &&
                (char.IsLetterOrDigit(text[index]) || text[index] == '\'' || text[index] == '-'))
            {
                index++;
                while (index < text.Length && text[index] <= 0x7f &&
                       (char.IsLetterOrDigit(text[index]) || text[index] == '\'' || text[index] == '-'))
                {
                    index++;
                }
                return text.Substring(start, index - start);
            }
            if (char.IsHighSurrogate(text[index]) && index + 1 < text.Length &&
                char.IsLowSurrogate(text[index + 1]))
            {
                index += 2;
                return text.Substring(start, 2);
            }
            index++;
            return text.Substring(start, 1);
        }

        private static int WrapCjkText(TextMeshProUGUI textElement, float maxWidth)
        {
            var input = textElement.text ?? string.Empty;
            if (input.Length == 0 || maxWidth <= 0f)
            {
                return 0;
            }

            textElement.ForceMeshUpdate(false, false);

            var result = new StringBuilder(input.Length + 16);
            var line = new StringBuilder();
            var visibleTokens = 0;
            var insertedBreaks = 0;
            var index = 0;
            while (index < input.Length)
            {
                var token = NextWrapToken(input, ref index, out var visible);
                if (token == "\n" || token == "\r" || token == "\r\n")
                {
                    result.Append(line);
                    result.Append(token);
                    line.Clear();
                    visibleTokens = 0;
                    continue;
                }

                var candidate = line.ToString() + token;
                var exceeds = visible && visibleTokens > 0 &&
                    textElement.GetPreferredValues(candidate).x > maxWidth;
                if (exceeds && !CannotBeginLine(token) && !CannotEndLine(line))
                {
                    while (line.Length > 0 && line[line.Length - 1] == ' ')
                    {
                        line.Length--;
                    }
                    result.Append(line);
                    result.Append('\n');
                    insertedBreaks++;
                    line.Clear();
                    visibleTokens = 0;
                    if (token == " ")
                    {
                        continue;
                    }
                }

                line.Append(token);
                if (visible && !string.IsNullOrWhiteSpace(token))
                {
                    visibleTokens++;
                }
            }
            result.Append(line);
            SetTextWithoutCapture(textElement, result.ToString());
            return insertedBreaks;
        }

        private static void OnSceneLoaded(Scene scene, LoadSceneMode mode)
        {
            RefreshAll($"scene:{scene.name}");
        }

        private IEnumerator RefreshLoop()
        {
            for (var attempt = 0; attempt < 20; attempt++)
            {
                yield return new WaitForSecondsRealtime(attempt < 5 ? 1f : 3f);
                RefreshAll($"startup:{attempt + 1}");
            }
        }

        private static void RefreshAll(string reason)
        {
            try
            {
                var dataChanges = PatchLoadedScriptableObjects();
                var textChanges = PatchLoadedTextComponents();
                if (dataChanges + textChanges > 0)
                {
                    Log.LogDebug($"{reason}：更新数据 {dataChanges} 项，界面文本 {textChanges} 项。");
                }
            }
            catch (Exception exception)
            {
                Log.LogWarning($"刷新汉化时发生异常（{reason}）：{exception}");
            }
        }

        private static int PatchLoadedTextComponents()
        {
            var changes = 0;
            if (!GlobalFallbackInstalled && EnsureGlobalChineseFallback())
            {
                Log.LogInfo("TMP 全局设置现已就绪，已补装中文 fallback 字体。");
            }
            foreach (var text in Resources.FindObjectsOfTypeAll<TMP_Text>())
            {
                if (text == null)
                {
                    continue;
                }
                AttachChineseFallback(text.font);
                if (IsDialogueTypewriterText(text))
                {
                    // Dialogue content is localised before the typewriter is
                    // started.  Replacing TMP.text again here races its
                    // maxVisibleCharacters animation: v1.2.16 tried healing
                    // idle typewriters here, misdetected mid-typing components
                    // as idle, cached half-revealed English as the source and
                    // rewrote the buffer mid-animation.  Never touch them.
                    continue;
                }
                var source = GetOriginalTextValue(text, text.text);
                var replacement = ChineseEnabled
                    ? (TryLookupTranslation(source, out var translated)
                        ? translated
                        : RelocalizeCompositeText(source, true))
                    : source;
                if (!string.Equals(text.text, replacement, StringComparison.Ordinal))
                {
                    if (ChineseEnabled)
                    {
                        ApplyChineseFontScale(text);
                    }
                    SetTextWithoutCapture(text, replacement);
                    changes++;
                }
            }

            foreach (var text in Resources.FindObjectsOfTypeAll<Text>())
            {
                if (text == null)
                {
                    continue;
                }
                var source = GetOriginalTextValue(text, text.text);
                var replacement = ChineseEnabled
                    ? (TryLookupTranslation(source, out var translated)
                        ? translated
                        : RelocalizeCompositeText(source, true))
                    : source;
                if (!string.Equals(text.text, replacement, StringComparison.Ordinal))
                {
                    SetTextWithoutCapture(text, replacement);
                    changes++;
                }
            }
            return changes;
        }

        private static bool IsDialogueTypewriterText(TMP_Text text)
        {
            return text != null &&
                text.GetComponent<Travelling.UI.Dialogue.TravellingTypewriter>() != null;
        }

        private static TMP_Text GetTypewriterTextComponent(
            Travelling.UI.Dialogue.TravellingTypewriter typewriter)
        {
            return typewriter == null
                ? null
                : GetMemberValue(typewriter, "m_textComponent") as TMP_Text ??
                    typewriter.GetComponent<TMP_Text>();
        }

        private static void RelocalizeVisibleTextComponents(bool toChinese)
        {
            foreach (var panel in Resources.FindObjectsOfTypeAll<Travelling.UI.Dialogue.TravellingSubtitlePanel>())
            {
                if (panel == null)
                {
                    continue;
                }
                var subtitle = GetMemberValue(panel, "currentSubtitle") ??
                    GetMemberValue(panel, "m_currentSubtitle");
                RelocalizeSubtitle(subtitle, toChinese);
            }
            foreach (var text in Resources.FindObjectsOfTypeAll<TMP_Text>())
            {
                if (text == null)
                {
                    continue;
                }
                if (IsDialogueTypewriterText(text))
                {
                    continue;
                }
                AttachChineseFallback(text.font);
                if (OriginalTextValues.ContainsKey(text))
                {
                    // Cached components have already been handled from their
                    // exact authored source. Avoid a second lossy/global pass.
                    continue;
                }
                var current = text.text;
                var replacement = RelocalizeCompositeText(current, toChinese);
                if (!string.Equals(current, replacement, StringComparison.Ordinal))
                {
                    if (toChinese)
                    {
                        // Some serialized scene/prefab text never passes
                        // through a setter before this scan. Preserve its
                        // component-specific authored English before the
                        // suppressed Chinese write so F9 can restore even a
                        // short or globally ambiguous translation exactly.
                        OriginalTextValues[text] = current;
                    }
                    SetTextWithoutCapture(text, replacement);
                }
            }
            foreach (var text in Resources.FindObjectsOfTypeAll<Text>())
            {
                if (text == null)
                {
                    continue;
                }
                if (OriginalTextValues.ContainsKey(text))
                {
                    continue;
                }
                var current = text.text;
                var replacement = RelocalizeCompositeText(current, toChinese);
                if (!string.Equals(current, replacement, StringComparison.Ordinal))
                {
                    if (toChinese)
                    {
                        OriginalTextValues[text] = current;
                    }
                    SetTextWithoutCapture(text, replacement);
                }
            }
        }

        private static void RelocalizeVisibleTypewriters(bool toChinese)
        {
            var handled = new HashSet<Travelling.UI.Dialogue.TravellingTypewriter>();
            foreach (var panel in Resources.FindObjectsOfTypeAll<Travelling.UI.Dialogue.TravellingSubtitlePanel>())
            {
                if (panel == null || !panel.isActiveAndEnabled || !panel.gameObject.activeInHierarchy)
                {
                    continue;
                }

                var typewriter = GetMemberValue(panel, "_typewriter") as
                        Travelling.UI.Dialogue.TravellingTypewriter ??
                    GetMemberValue(panel, "Typewriter") as
                        Travelling.UI.Dialogue.TravellingTypewriter;
                if (typewriter == null || !handled.Add(typewriter) ||
                    !typewriter.isActiveAndEnabled || !typewriter.gameObject.activeInHierarchy)
                {
                    continue;
                }
                var text = GetTypewriterTextComponent(typewriter);
                if (text == null || !text.gameObject.activeInHierarchy)
                {
                    continue;
                }
                AttachChineseFallback(text.font);

                var current = text.text;
                if (!OriginalTextValues.TryGetValue(text, out var source))
                {
                    // TravellingSubtitlePanel accumulates the entire visible
                    // dialogue history in this one TMP component.  Never fall
                    // back to currentSubtitle here: doing so replaces that
                    // accumulated buffer with only the newest line.  An older
                    // plugin build may not yet have captured the component, so
                    // reverse the complete rendered buffer once if possible.
                    source = ContainsCjk(current)
                        ? RelocalizeDialogueBufferSource(current)
                        : current;
                    if (ContainsCjk(source))
                    {
                        // Lossless reversal is impossible. Leave the complete
                        // history in place instead of destructively truncating it.
                        continue;
                    }
                    OriginalTextValues[text] = source;
                }
                var replacement = toChinese
                    ? LocalizeDialogueBufferChinese(source)
                    : source;
                if (string.Equals(current, replacement, StringComparison.Ordinal))
                {
                    continue;
                }

                var playingValue = GetMemberValue(typewriter, "isPlaying") ??
                    GetMemberValue(typewriter, "IsPlaying");
                var wasPlaying = playingValue is bool flag && flag;
                if (wasPlaying)
                {
                    // RestartFromIndexPreservingState retains cadence, pause and
                    // click state. Its index is a TMP visible-character count,
                    // not a backing-string offset, so rich tags remain safe.
                    text.ForceMeshUpdate(false, false);
                    var oldVisibleCount = text.textInfo?.characterCount ?? 0;
                    var visibleCharacters = Mathf.Clamp(
                        text.maxVisibleCharacters,
                        0,
                        oldVisibleCount);
                    typewriter.RestartFromIndexPreservingState(replacement, visibleCharacters);
                }
                else
                {
                    // A completed subtitle must remain completed. Restarting its
                    // coroutine would replay onBegin/onEnd, Typed, and scrolling.
                    SetTextWithoutCapture(text, replacement);
                    text.ForceMeshUpdate(false, false);
                    text.maxVisibleCharacters = text.textInfo?.characterCount ?? 0;
                }
            }
        }

        private static void RefreshVisibleWorldPopups()
        {
            var composeMethod = AccessTools.Method(
                typeof(Travelling.Interactables.WorldPopup),
                "ComposeDisplayText",
                Type.EmptyTypes);
            var styleMethod = AccessTools.Method(
                typeof(Travelling.Interactables.WorldPopup),
                "ApplyPaperStripStyle",
                Type.EmptyTypes);
            if (composeMethod == null)
            {
                return;
            }

            foreach (var popup in Resources.FindObjectsOfTypeAll<Travelling.Interactables.WorldPopup>())
            {
                if (popup == null || !popup.isActiveAndEnabled || !popup.gameObject.activeInHierarchy ||
                    !popup.IsVisible)
                {
                    continue;
                }
                var text = GetMemberValue(popup, "plaqueText") as TMP_Text;
                if (text == null)
                {
                    continue;
                }
                try
                {
                    var rendered = composeMethod.Invoke(popup, null) as string ?? string.Empty;
                    SetTextWithoutCapture(text, rendered);
                    text.ForceMeshUpdate(false, false);
                    styleMethod?.Invoke(popup, null);
                    var parent = text.rectTransform.parent as RectTransform;
                    if (parent != null)
                    {
                        LayoutRebuilder.ForceRebuildLayoutImmediate(parent);
                    }
                }
                catch (Exception exception)
                {
                    Log.LogWarning($"切换语言后重排场景互动提示失败：{exception.Message}");
                }
            }
        }

        [HarmonyPatch(
            typeof(Travelling.UI.Dialogue.TravellingSubtitlePanel),
            "SetSubtitleTextContent")]
        private static class SubtitleBufferSourcePatch
        {
            private static void Prefix(object subtitle)
            {
                // The game computes the typewriter start index from the old
                // accumulated buffer before it calls TravellingTypewriter. The
                // current subtitle must therefore already be in the selected
                // language here; translating only at the typewriter boundary
                // makes an English glyph index skip over shorter Chinese text.
                DialogueBufferBuildDepth++;
                RelocalizeSubtitle(subtitle, ChineseEnabled);
            }

            private static Exception Finalizer(Exception __exception)
            {
                DialogueBufferBuildDepth = Math.Max(0, DialogueBufferBuildDepth - 1);
                return __exception;
            }
        }

        private static string GetOriginalTextValue(object owner, string current)
        {
            if (owner == null)
            {
                return current;
            }
            if (OriginalTextValues.TryGetValue(owner, out var source))
            {
                return source;
            }
            OriginalTextValues[owner] = current;
            return current;
        }

        private static void ApplyChineseFontScale(TMP_Text text)
        {
            if (text == null || ChineseFontScale <= 1f)
            {
                return;
            }
            if (!OriginalFontSizes.TryGetValue(text, out var original))
            {
                original = text.fontSize;
                OriginalFontSizes[text] = original;
            }
            // Auto-size already reacts to the available rectangle.  Applying
            // a second multiplier there would increase clipping risk.
            if (!text.enableAutoSizing)
            {
                text.fontSize = original * ChineseFontScale;
            }
        }

        private static void SetTextWithoutCapture(TMP_Text text, string value)
        {
            if (text == null)
            {
                return;
            }
            try
            {
                TextWriteSuppressionDepth++;
                text.text = value;
            }
            finally
            {
                TextWriteSuppressionDepth--;
            }
        }

        private static void SetTextWithoutCapture(Text text, string value)
        {
            if (text == null)
            {
                return;
            }
            try
            {
                TextWriteSuppressionDepth++;
                text.text = value;
            }
            finally
            {
                TextWriteSuppressionDepth--;
            }
        }

        private static void AttachChineseFallback(TMP_FontAsset font)
        {
            if (font == null || ChineseFont == null || font == ChineseFont)
            {
                return;
            }
            var fallbacks = font.fallbackFontAssetTable;
            if (fallbacks == null)
            {
                fallbacks = new List<TMP_FontAsset>();
                font.fallbackFontAssetTable = fallbacks;
            }
            if (!fallbacks.Contains(ChineseFont))
            {
                fallbacks.Insert(0, ChineseFont);
            }
        }

        private static int PatchLoadedScriptableObjects()
        {
            var changes = 0;
            var curators = new List<ScriptableObject>();
            foreach (var root in Resources.FindObjectsOfTypeAll<ScriptableObject>())
            {
                if (root == null)
                {
                    continue;
                }
                var fullName = root.GetType().FullName ?? string.Empty;
                if (string.Equals(
                    fullName,
                    "Travelling.PCQualities.ScriptablesCurator",
                    StringComparison.Ordinal))
                {
                    curators.Add(root);
                    TrackedCurators[root] = true;
                }
                if (fullName.IndexOf("DialogueDatabase", StringComparison.Ordinal) >= 0)
                {
                    changes += PatchDialogueDatabase(root);
                    continue;
                }
                if (!fullName.StartsWith("Travelling.", StringComparison.Ordinal) &&
                    !string.Equals(root.GetType().Name, "LocData", StringComparison.Ordinal))
                {
                    continue;
                }

                if (string.Equals(root.GetType().Name, "LocData", StringComparison.Ordinal))
                {
                    TrackOriginalLocTemplates(root);
                }

                var visited = new HashSet<object>(ReferenceEqualityComparer.Instance);
                changes += PatchStructuredObject(root, root.GetType(), visited, 0);
            }
            if (changes > 0)
            {
                RefreshScriptableCuratorCaches(curators);
            }
            return changes;
        }

        private static void TrackOriginalLocTemplates(object locData)
        {
            foreach (var entry in EnumerateMember(locData, "entries"))
            {
                var key = GetMemberValue(entry, "key") as string;
                var current = GetMemberValue(entry, "value") as string;
                if (string.IsNullOrEmpty(key) || current == null)
                {
                    continue;
                }
                var source = GetOriginalMemberValue(entry, "value", current);
                OriginalLocTemplatesByKey[key] = source;
            }
        }

        private static void RefreshScriptableCuratorCaches(IEnumerable<ScriptableObject> curators)
        {
            foreach (var curator in curators)
            {
                try
                {
                    var method = curator.GetType().GetMethod("ForceRefresh", AnyInstance);
                    if (method == null || method.GetParameters().Length != 0)
                    {
                        Log.LogWarning("未找到 ScriptablesCurator.ForceRefresh()；标签缓存可能仍引用英文键。");
                        continue;
                    }
                    method.Invoke(curator, null);
                }
                catch (Exception exception)
                {
                    Log.LogWarning($"刷新 ScriptablesCurator 标签/别名缓存失败：{exception}");
                }
            }
        }

        private static int PatchDialogueDatabase(object database)
        {
            var changes = 0;
            foreach (var record in EnumerateMember(database, "actors"))
            {
                changes += PatchDialogueFields(record);
            }
            foreach (var record in EnumerateMember(database, "items"))
            {
                changes += PatchDialogueFields(record);
            }
            foreach (var conversation in EnumerateMember(database, "conversations"))
            {
                changes += PatchDialogueFields(conversation);
                foreach (var entry in EnumerateMember(conversation, "dialogueEntries"))
                {
                    changes += PatchDialogueFields(entry);
                }
            }
            return changes;
        }

        private static int PatchDialogueFields(object record)
        {
            var changes = 0;
            foreach (var field in EnumerateMember(record, "fields"))
            {
                var title = GetMemberValue(field, "title") as string ?? string.Empty;
                var allowed = DialogueFieldNames.Contains(title) ||
                    (title.StartsWith("SkillCheckModifier_", StringComparison.Ordinal) &&
                     title.EndsWith("_Description", StringComparison.Ordinal));
                if (!allowed)
                {
                    continue;
                }
                var source = GetMemberValue(field, "value") as string;
                // Keep authored dialogue data in English. UI assignment hooks
                // localise at the rendering boundary, preserving a lossless
                // source for dynamic controls and fast F9 restoration.
                TryLookupTranslation(source, out _);
            }
            return changes;
        }

        private static int PatchStructuredObject(
            object value,
            Type rootType,
            HashSet<object> visited,
            int depth)
        {
            if (value == null || depth > 6 || value is string || value.GetType().IsPrimitive)
            {
                return 0;
            }
            if (!value.GetType().IsValueType && !visited.Add(value))
            {
                return 0;
            }

            var changes = 0;
            foreach (var field in GetAllFields(value.GetType()))
            {
                if (field.IsStatic)
                {
                    continue;
                }

                object child;
                try
                {
                    child = field.GetValue(value);
                }
                catch
                {
                    continue;
                }

                if (field.FieldType == typeof(string))
                {
                    var allowed = TranslatableFieldNames.Contains(field.Name) ||
                        (field.Name.Equals("text", StringComparison.OrdinalIgnoreCase) &&
                         string.Equals(
                             field.DeclaringType?.FullName,
                             "Travelling.PCQualities.TransitionMessageEntry",
                             StringComparison.Ordinal)) ||
                        (field.Name.Equals("value", StringComparison.OrdinalIgnoreCase) &&
                         rootType.Name.Equals("LocData", StringComparison.Ordinal));
                    if (!allowed || field.IsInitOnly)
                    {
                        continue;
                    }
                    var source = child as string;
                    var isLabel = field.Name.Equals("label", StringComparison.OrdinalIgnoreCase) ||
                        field.Name.Equals("_label", StringComparison.OrdinalIgnoreCase);
                    if (isLabel && !string.IsNullOrEmpty(source))
                    {
                        // Link literals are visible text as well as lookup keys.  Register
                        // reviewed aliases even when a proper-name label intentionally stays
                        // unchanged and therefore has no direct translation mutation.
                        PreserveSourceLabelAlias(value, source);
                    }
                    TryLookupTranslation(source, out _);
                    continue;
                }

                if (child is IList list &&
                    TranslatableStringCollectionFieldNames.Contains(field.Name))
                {
                    for (var index = 0; index < list.Count; index++)
                    {
                        var source = list[index] as string;
                        TryLookupTranslation(source, out _);
                    }
                    continue;
                }

                if (child is UnityEngine.Object)
                {
                    continue;
                }
                if (child is IEnumerable enumerable)
                {
                    foreach (var item in enumerable)
                    {
                        changes += PatchStructuredObject(item, rootType, visited, depth + 1);
                    }
                }
                else if (!field.FieldType.IsPrimitive && !field.FieldType.IsEnum)
                {
                    changes += PatchStructuredObject(child, rootType, visited, depth + 1);
                }
            }
            return changes;
        }

        private static string GetOriginalMemberValue(object owner, string name, string current)
        {
            if (owner == null || string.IsNullOrEmpty(name))
            {
                return current;
            }
            if (!OriginalMemberValues.TryGetValue(owner, out var members))
            {
                members = new Dictionary<string, string>(StringComparer.Ordinal);
                OriginalMemberValues[owner] = members;
            }
            if (members.TryGetValue(name, out var source))
            {
                if (!string.Equals(current, source, StringComparison.Ordinal) &&
                    !IsRenderedLocalization(source, current))
                {
                    members[name] = current;
                    return current;
                }
                return source;
            }
            members[name] = current;
            return current;
        }

        private static string GetOriginalListValue(object owner, int index, string current)
        {
            if (owner == null)
            {
                return current;
            }
            if (!OriginalListValues.TryGetValue(owner, out var values))
            {
                values = new Dictionary<int, string>();
                OriginalListValues[owner] = values;
            }
            if (values.TryGetValue(index, out var source))
            {
                if (!string.Equals(current, source, StringComparison.Ordinal) &&
                    !IsRenderedLocalization(source, current))
                {
                    values[index] = current;
                    return current;
                }
                return source;
            }
            values[index] = current;
            return current;
        }

        private static void RestoreOriginalValues()
        {
            var restored = 0;
            var structuredDataChanged = false;
            foreach (var ownerEntry in OriginalMemberValues.ToArray())
            {
                if (IsDestroyedUnityObject(ownerEntry.Key))
                {
                    OriginalMemberValues.Remove(ownerEntry.Key);
                    continue;
                }
                foreach (var member in ownerEntry.Value)
                {
                    var current = GetMemberValue(ownerEntry.Key, member.Key) as string;
                    if (!string.Equals(current, member.Value, StringComparison.Ordinal) &&
                        SetMemberValue(ownerEntry.Key, member.Key, member.Value))
                    {
                        restored++;
                        structuredDataChanged = true;
                    }
                }
            }
            foreach (var ownerEntry in OriginalListValues.ToArray())
            {
                if (ownerEntry.Key is IList list)
                {
                    foreach (var value in ownerEntry.Value)
                    {
                        if (value.Key < 0 || value.Key >= list.Count)
                        {
                            continue;
                        }
                        try
                        {
                            if (!string.Equals(list[value.Key] as string, value.Value, StringComparison.Ordinal))
                            {
                                list[value.Key] = value.Value;
                                restored++;
                                structuredDataChanged = true;
                            }
                        }
                        catch (NotSupportedException)
                        {
                            break;
                        }
                    }
                }
            }
            foreach (var entry in OriginalTextValues.ToArray())
            {
                if (entry.Key is TMP_Text tmp && tmp != null)
                {
                    // Dialogue typewriters hold the complete accumulated
                    // history. RelocalizeVisibleTypewriters owns these writes
                    // so it can preserve the current reveal index as well.
                    if (IsDialogueTypewriterText(tmp))
                    {
                        continue;
                    }
                    if (!string.Equals(tmp.text, entry.Value, StringComparison.Ordinal))
                    {
                        SetTextWithoutCapture(tmp, entry.Value);
                        restored++;
                    }
                }
                else if (entry.Key is Text legacy && legacy != null)
                {
                    if (!string.Equals(legacy.text, entry.Value, StringComparison.Ordinal))
                    {
                        SetTextWithoutCapture(legacy, entry.Value);
                        restored++;
                    }
                }
                else if (IsDestroyedUnityObject(entry.Key))
                {
                    OriginalTextValues.Remove(entry.Key);
                }
            }
            foreach (var entry in OriginalFontSizes.ToArray())
            {
                if (entry.Key is TMP_Text tmp && tmp != null)
                {
                    if (!Mathf.Approximately(tmp.fontSize, entry.Value))
                    {
                        tmp.fontSize = entry.Value;
                    }
                }
                else if (IsDestroyedUnityObject(entry.Key))
                {
                    OriginalFontSizes.Remove(entry.Key);
                }
            }
            if (structuredDataChanged)
            {
                RefreshScriptableCuratorCaches(GetLiveTrackedCurators());
            }
            Log.LogDebug($"切换至英文：恢复 {restored} 个原始值。");
        }

        private static void ApplyCachedLanguageValues(bool toChinese)
        {
            if (!toChinese)
            {
                RestoreOriginalValues();
                return;
            }
            var changed = 0;
            // Data objects remain in authored English. Register their cached
            // sources so rendering-boundary hooks can translate new UI.
            foreach (var ownerEntry in OriginalMemberValues.ToArray())
            {
                if (IsDestroyedUnityObject(ownerEntry.Key))
                {
                    OriginalMemberValues.Remove(ownerEntry.Key);
                    continue;
                }
                foreach (var member in ownerEntry.Value)
                {
                    TryLookupTranslation(member.Value, out _);
                }
            }
            foreach (var ownerEntry in OriginalListValues.ToArray())
            {
                if (!(ownerEntry.Key is IList list))
                {
                    OriginalListValues.Remove(ownerEntry.Key);
                    continue;
                }
                foreach (var value in ownerEntry.Value)
                {
                    TryLookupTranslation(value.Value, out _);
                }
            }
            // Register every exact cached source first so composite rendering
            // below is independent of dictionary traversal order.
            foreach (var entry in OriginalTextValues.ToArray())
            {
                if (!IsDestroyedUnityObject(entry.Key))
                {
                    TryLookupTranslation(entry.Value, out _);
                }
            }
            foreach (var entry in OriginalTextValues.ToArray())
            {
                if (IsDestroyedUnityObject(entry.Key))
                {
                    OriginalTextValues.Remove(entry.Key);
                    continue;
                }
                var replacement = TryLookupTranslation(entry.Value, out var translated)
                    ? translated
                    : RelocalizeCompositeText(entry.Value, true);
                if (entry.Key is TMP_Text tmp && tmp != null)
                {
                    if (IsDialogueTypewriterText(tmp))
                    {
                        continue;
                    }
                    AttachChineseFallback(tmp.font);
                    ApplyChineseFontScale(tmp);
                    if (!string.Equals(tmp.text, replacement, StringComparison.Ordinal))
                    {
                        SetTextWithoutCapture(tmp, replacement);
                        changed++;
                    }
                }
                else if (entry.Key is Text legacy && legacy != null)
                {
                    if (!string.Equals(legacy.text, replacement, StringComparison.Ordinal))
                    {
                        SetTextWithoutCapture(legacy, replacement);
                        changed++;
                    }
                }
            }
            Log.LogDebug($"切换至中文：从缓存更新 {changed} 个值。");
        }

        private static bool IsDestroyedUnityObject(object value)
        {
            return value is UnityEngine.Object unityObject && unityObject == null;
        }

        private static void PreserveSourceLabelAlias(object owner, string source)
        {
            if (owner == null || string.IsNullOrEmpty(source))
            {
                return;
            }
            var aliases = GetMemberValue(owner, "alternativeLabels") as IList ??
                GetMemberValue(owner, "_alternativeLabels") as IList;
            if (aliases == null)
            {
                var newAliases = new List<string>();
                if (!SetMemberValue(owner, "alternativeLabels", newAliases) &&
                    !SetMemberValue(owner, "_alternativeLabels", newAliases))
                {
                    return;
                }
                aliases = newAliases;
            }
            var sourceAliases = new List<string> { source };
            foreach (var alias in aliases)
            {
                if (alias is string aliasText && !string.IsNullOrEmpty(aliasText))
                {
                    sourceAliases.Add(aliasText);
                }
            }
            foreach (var alias in sourceAliases)
            {
                AddAliasIfMissing(aliases, alias);
                if (LinkTargetTranslations.TryGetValue(GetSourceHash(alias), out var translatedAlias))
                {
                    AddAliasIfMissing(aliases, translatedAlias);
                }
            }
        }

        private static void AddAliasIfMissing(IList aliases, string alias)
        {
            if (string.IsNullOrEmpty(alias))
            {
                return;
            }
            foreach (var existing in aliases)
            {
                if (string.Equals(existing as string, alias, StringComparison.OrdinalIgnoreCase))
                {
                    return;
                }
            }
            try
            {
                aliases.Add(alias);
            }
            catch (NotSupportedException)
            {
                // Fixed-size arrays cannot accept aliases; leave them unchanged.
            }
        }

        private static IEnumerable<FieldInfo> GetAllFields(Type type)
        {
            while (type != null && type != typeof(object))
            {
                foreach (var field in type.GetFields(AnyInstance | BindingFlags.DeclaredOnly))
                {
                    yield return field;
                }
                type = type.BaseType;
            }
        }

        private static IEnumerable<object> EnumerateMember(object owner, string name)
        {
            if (GetMemberValue(owner, name) is IEnumerable enumerable)
            {
                foreach (var item in enumerable)
                {
                    if (item != null)
                    {
                        yield return item;
                    }
                }
            }
        }

        private static object GetMemberValue(object owner, string name)
        {
            if (owner == null)
            {
                return null;
            }
            var type = owner.GetType();
            while (type != null)
            {
                var field = type.GetField(name, AnyInstance | BindingFlags.DeclaredOnly);
                if (field != null)
                {
                    return field.GetValue(owner);
                }
                var property = type.GetProperty(name, AnyInstance | BindingFlags.DeclaredOnly);
                if (property != null && property.CanRead)
                {
                    return property.GetValue(owner, null);
                }
                type = type.BaseType;
            }
            return null;
        }

        private static bool SetMemberValue(object owner, string name, object value)
        {
            if (owner == null)
            {
                return false;
            }
            var type = owner.GetType();
            while (type != null)
            {
                var field = type.GetField(name, AnyInstance | BindingFlags.DeclaredOnly);
                if (field != null && !field.IsInitOnly)
                {
                    field.SetValue(owner, value);
                    return true;
                }
                var property = type.GetProperty(name, AnyInstance | BindingFlags.DeclaredOnly);
                if (property != null && property.CanWrite)
                {
                    property.SetValue(owner, value, null);
                    return true;
                }
                type = type.BaseType;
            }
            return false;
        }

        private static void LocalizeTmpAssignment(TMP_Text instance, ref string value)
        {
            if (TextWriteSuppressionDepth > 0 || instance == null || IsDialogueTypewriterText(instance))
            {
                return;
            }
            AttachChineseFallback(instance.font);
            if (!ChineseEnabled)
            {
                var reversed = RelocalizeCompositeText(value, false);
                if (OriginalTextValues.TryGetValue(instance, out var original))
                {
                    if (IsRenderedLocalization(original, value))
                    {
                        value = original;
                        return;
                    }
                    if (!string.Equals(reversed, value, StringComparison.Ordinal))
                    {
                        value = reversed;
                        OriginalTextValues[instance] = reversed;
                        return;
                    }
                    if (ContainsCjk(value))
                    {
                        // A Chinese backing field attempted to repaint after
                        // F9. Never let it poison the authored English cache.
                        value = original;
                        return;
                    }
                    OriginalTextValues[instance] = value;
                }
                else
                {
                    value = reversed;
                    if (ContainsCjk(value))
                    {
                        // Ambiguous Chinese cannot be reversed losslessly.
                        // Leave it uncached so a later authored assignment can
                        // establish the correct component-specific source.
                        return;
                    }
                    OriginalTextValues[instance] = value;
                }
                return;
            }
            if (!OriginalTextValues.TryGetValue(instance, out var source) ||
                string.Equals(value, source, StringComparison.Ordinal) ||
                (!IsKnownTranslation(source, value) &&
                 !string.Equals(value, instance.text, StringComparison.Ordinal)))
            {
                source = value;
                if (ContainsCjk(source))
                {
                    var reversed = RelocalizeCompositeText(source, false);
                    if (string.Equals(reversed, source, StringComparison.Ordinal))
                    {
                        // Runtime formatting can combine a Chinese template with
                        // English internal tokens (for example, "Dignify ...
                        // Trace"). Translate the known visible tokens, but never
                        // cache this mixed rendered value as the English source.
                        var mixedReplacement = RelocalizeCompositeText(source, true);
                        if (!string.Equals(mixedReplacement, source, StringComparison.Ordinal))
                        {
                            ApplyChineseFontScale(instance);
                            value = mixedReplacement;
                        }
                        return;
                    }
                    source = reversed;
                }
                OriginalTextValues[instance] = source;
            }
            if (TryTranslate(source, out var translated))
            {
                ApplyChineseFontScale(instance);
                value = translated;
            }
            else
            {
                var replacement = RelocalizeCompositeText(source, true);
                if (!string.Equals(source, replacement, StringComparison.Ordinal))
                {
                    ApplyChineseFontScale(instance);
                    value = replacement;
                }
                else if (DiagLooksUntranslated(source))
                {
                    DiagRecord("TMP", "miss", source);
                }
            }
        }

        private static bool IsRenderedLocalization(string source, string value)
        {
            if (string.IsNullOrEmpty(source) || string.IsNullOrEmpty(value))
            {
                return false;
            }
            if (IsKnownTranslation(source, value))
            {
                return true;
            }
            return string.Equals(
                RelocalizeCompositeText(source, true),
                value,
                StringComparison.Ordinal);
        }

        [HarmonyPatch]
        private static class DialogueTypewriterInputPatch
        {
            private static IEnumerable<MethodBase> TargetMethods()
            {
                var type = typeof(Travelling.UI.Dialogue.TravellingTypewriter);
                return new[] { "StartTyping", "PlayText", "RestartFromIndexPreservingState" }
                    .Select(name => AccessTools.Method(type, name, new[] { typeof(string), typeof(int) }))
                    .Where(method => method != null);
            }

            private static void Prefix(
                Travelling.UI.Dialogue.TravellingTypewriter __instance,
                ref string __0)
            {
                var text = GetTypewriterTextComponent(__instance);
                var source = __0 ?? string.Empty;
                if (text != null)
                {
                    if (OriginalTextValues.TryGetValue(text, out var savedSource) &&
                        (string.Equals(source, savedSource, StringComparison.Ordinal) ||
                         IsRenderedLocalization(savedSource, source)))
                    {
                        source = savedSource;
                    }
                    else
                    {
                        // StartTyping receives the panel's complete accumulated
                        // buffer (dimmed history plus the new line), not merely
                        // currentSubtitle. Cache that full English buffer so F9
                        // can switch every existing line without rebuilding UI.
                        var reversed = RelocalizeDialogueBufferSource(source);
                        if (ContainsCjk(reversed))
                        {
                            // 反向还原残留中文：这条缓冲被缓存为"英文源"后会污染
                            // F9 还原，诊断采集其确切形态。
                            DiagRecord("BufferSource", "cjk", reversed);
                        }
                        // Always advance the cache to the newest complete
                        // buffer. Falling back to savedSource here discards the
                        // newly appended line whenever it is a short Chinese
                        // utterance, which is why that line vanished when the
                        // delayed typewriter restart ran.
                        source = reversed;
                        OriginalTextValues[text] = source;
                    }
                }
                var replacement = ChineseEnabled
                    ? LocalizeDialogueBufferChinese(source)
                    : source;
                if (ChineseEnabled &&
                    !string.Equals(source, replacement, StringComparison.Ordinal))
                {
                    // The next accumulated buffer embeds this exact rendered
                    // history. Register it as one reversible unit so later
                    // short lines do not depend on lossy global substitution.
                    RegisterKnownLocalizationPair(source, replacement);
                }
                // v1.2.16 曾在此按可视字符重映射起始索引 __1：中英文本长度差
                // 的推断离不了真机验证，且与自愈回写叠加后导致"从很多句之前
                // 重新打字"。回退为透传——根治方案是把译文烘进游戏资产，
                // 让游戏用原生中文计算索引，而不是在渲染边界修修补补。
                __0 = replacement;
                if (ChineseEnabled)
                {
                    foreach (var line in replacement.Split('\n'))
                    {
                        var trimmedLine = line.Trim('\r');
                        if (DiagLooksUntranslated(trimmedLine))
                        {
                            DiagRecord("Typewriter", "miss", trimmedLine);
                        }
                    }
                }
            }
        }

        [HarmonyPatch]
        private static class LocFormattedTemplatePatch
        {
            private static IEnumerable<MethodBase> TargetMethods()
            {
                var locType = AccessTools.TypeByName("Travelling.Loc");
                if (locType == null)
                {
                    return Enumerable.Empty<MethodBase>();
                }
                return locType
                    .GetMethods(BindingFlags.Static | BindingFlags.Public)
                    .Where(method =>
                        method.Name == "ForCurrentCulture" &&
                        method.ReturnType == typeof(string) &&
                        method.GetParameters().Length == 2 &&
                        method.GetParameters()[0].ParameterType == typeof(string) &&
                        method.GetParameters()[1].ParameterType == typeof(string[]))
                    .Cast<MethodBase>();
            }

            private static void Postfix(string __0, string[] __1, ref string __result)
            {
                // Loc formats dynamic display arguments after choosing the
                // template. Translate the completed visible string here so
                // advice panels retain a deterministic per-language value,
                // rather than relying on a later TMP repaint of mixed text.
                if (ChineseEnabled)
                {
                    var renderedValue = __result;
                    var replacement = RelocalizeCompositeText(renderedValue, true);
                    if (OriginalLocTemplatesByKey.TryGetValue(__0, out var template) &&
                        TryLookupTranslation(template, out var translatedTemplate))
                    {
                        // Translate the template before formatting it. Replacing
                        // only its expanded arguments leaves an English shell
                        // ("Failing 修养 checks will add 疲惫."), and can race a
                        // later UI repaint that still owns the formatted value.
                        var translatedArguments = (__1 ?? Array.Empty<string>())
                            .Select(LocalizeFormattedArgument)
                            .Cast<object>()
                            .ToArray();
                        try
                        {
                            replacement = string.Format(translatedTemplate, translatedArguments);
                        }
                        catch (FormatException exception)
                        {
                            // A malformed external template must not break the
                            // game's UI. Fall back to conservative token-level
                            // localisation and keep a diagnostic for review.
                            Log.LogWarning($"本地化格式模板无效（{__0}）：{exception.Message}");
                        }
                    }
                    if (!string.Equals(renderedValue, replacement, StringComparison.Ordinal))
                    {
                        // Bind exactly what Loc rendered, not a hand-built
                        // approximation of it. This keeps repaint and F9
                        // restoration stable even when an argument alias is
                        // many-to-one (for example, Dignify / Dignity).
                        RegisterKnownLocalizationPair(renderedValue, replacement);
                        __result = replacement;
                    }
                    else if (DiagLooksUntranslated(renderedValue))
                    {
                        DiagRecord("LocFmt", "miss", renderedValue);
                    }
                }
            }
        }
